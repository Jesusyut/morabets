function americanToDecimal(odds) {
  if (odds > 0) {
    return (odds / 100) + 1;
  } else {
    return (100 / Math.abs(odds)) + 1;
  }
}

function impliedProb(odds) {
  if (odds < 0) {
    return Math.abs(odds) / 
      (Math.abs(odds) + 100);
  } else {
    return 100 / (odds + 100);
  }
}

function noVigProb(odds) {
  // Single leg no-vig
  // For a single side we treat
  // the implied prob as approximate
  // true probability
  // (full no-vig requires both sides
  //  but for parlay calc use implied)
  return impliedProb(odds);
}

function decimalToAmerican(decimal) {
  if (decimal >= 2) {
    return Math.round((decimal - 1) * 100);
  } else {
    return Math.round(
      -100 / (decimal - 1)
    );
  }
}

function calculateParlay() {
  const wager = parseFloat(
    document.getElementById('wager').value
  ) || 100;
  
  const inputs = document.querySelectorAll(
    '.leg-odds-input'
  );
  
  let parlayDecimal = 1;
  let trueDecimal = 1;
  const legResults = [];
  
  for (let input of inputs) {
    const odds = parseFloat(input.value);
    if (isNaN(odds)) return;
    
    const decimal = americanToDecimal(odds);
    const trueProb = noVigProb(odds);
    
    parlayDecimal *= decimal;
    trueDecimal *= trueProb;
    
    legResults.push({ odds, trueProb });
  }
  
  const payout = wager * parlayDecimal;
  const profit = payout - wager;
  const parlayOdds = decimalToAmerican(
    parlayDecimal
  );
  const bookImplied = 1 / parlayDecimal;
  const edgePct = (
    (trueDecimal - bookImplied) * 100
  ).toFixed(2);
  const trueProb = (trueDecimal * 100)
    .toFixed(1);
  const bookProb = (bookImplied * 100)
    .toFixed(1);
  
  // Display results
  document.getElementById('result-odds')
    .textContent = parlayOdds > 0 
      ? '+' + parlayOdds 
      : parlayOdds;
  document.getElementById('result-payout')
    .textContent = '$' + payout.toFixed(2);
  document.getElementById('result-trueprob')
    .textContent = trueProb + '%';
  
  const edgeEl = document.getElementById(
    'result-edge'
  );
  edgeEl.textContent = (edgePct > 0 
    ? '+' : '') + edgePct + '%';
  edgeEl.style.color = edgePct > 0 
    ? '#4CBB17' : '#EF4444';
  
  document.getElementById('result-warning')
    .style.display = edgePct < 0 
      ? 'block' : 'none';
  
  // Leg breakdown
  const breakdown = document.getElementById(
    'leg-breakdown'
  );
  breakdown.innerHTML = legResults.map(
    (leg, i) => `
    <div style="display:flex;
                justify-content:space-between;
                padding:4px 0;
                border-bottom:1px solid 
                rgba(255,255,255,0.05);">
      <span style="color:#6B9E5A;
                   font-size:13px;">
        Leg ${i+1}: ${leg.odds > 0 
          ? '+' + leg.odds 
          : leg.odds}
      </span>
      <span style="color:#ffffff;
                   font-size:13px;">
        ${(leg.trueProb * 100).toFixed(1)}% 
        true prob
      </span>
    </div>
  `).join('');
  
  document.getElementById('combined-true')
    .textContent = trueProb + '%';
  document.getElementById('combined-book')
    .textContent = bookProb + '%';
  
  // Show results
  document.getElementById('results-section')
    .style.display = 'block';

  // Show Bovada CTA if parlay has edge
  document.getElementById('bovada-cta')
    .style.display = edgePct > 0 
      ? 'block' : 'none';
}

// Add leg button
function addLeg() {
  const container = document.getElementById(
    'legs-container'
  );
  const count = container.children.length + 1;
  if (count > 6) return;
  
  const div = document.createElement('div');
  div.className = 'leg-row';
  div.style.cssText = 'display:flex;' +
    'align-items:center;gap:8px;' +
    'margin-bottom:8px;';
  div.innerHTML = `
    <span style="color:#6B9E5A;
                 font-size:13px;
                 font-weight:600;
                 min-width:48px;">
      Leg ${count}
    </span>
    <input type="number"
           class="leg-odds-input"
           placeholder="-110"
           style="flex:1;border:1.5px solid 
                  #E5E7EB;border-radius:10px;
                  padding:12px;font-size:15px;">
    <button onclick="removeLeg(this)"
            style="color:#9CA3AF;
                   background:none;
                   border:none;
                   font-size:18px;
                   cursor:pointer;">
      ×
    </button>
  `;
  container.appendChild(div);
}

function removeLeg(btn) {
  const container = document.getElementById(
    'legs-container'
  );
  if (container.children.length <= 2) return;
  btn.parentElement.remove();
  // Renumber legs
  const rows = container.querySelectorAll(
    '.leg-row span'
  );
  rows.forEach((s, i) => {
    s.textContent = 'Leg ' + (i + 1);
  });
}

// Auto-calculate on input change
document.addEventListener('input', function(e) {
  if (e.target.classList.contains(
    'leg-odds-input'
  ) || e.target.id === 'wager') {
    calculateParlay();
  }
});

H2 Barlow Condensed Black:
"HOW TO READ YOUR PARLAY RESULTS"

Body Inter 16px:

The true probability is what the math
says your parlay should hit — with the
sportsbook's margin removed.

The book implied probability is what
the payout assumes you will hit. It is
always lower because the book takes
a margin on every leg.

When your true probability is higher
than the book implied probability you
have a positive edge. That means the
payout is mathematically favorable
relative to the real chance of winning.

Most parlays have negative edge.
That is by design.

Parlays built from individually sharp
lines — props where the no-vig
probability is strong on each leg —
are the only parlays worth considering.

H3:
"Why Most Parlays Lose"

The house edge on a standard -110/-110
straight bet is about 4.5%.

A two-leg parlay compounds that.
A four-leg parlay compounds it four times.

The only time a parlay has positive edge
is when you find two or more legs where
the true probability on each leg exceeds
what the book is pricing.

That is rare. When it exists it is worth
knowing. When it does not exist you are
just compounding the house take.
