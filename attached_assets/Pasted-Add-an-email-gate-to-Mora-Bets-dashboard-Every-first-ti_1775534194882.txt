=== TASK 2: EMAIL GATE JAVASCRIPT ===

Add this script at the bottom of 
dashboard.html before </body>:

<script>
(function() {

  // Check if already submitted this browser
  const hasAccess = localStorage.getItem(
    'mora_access'
  );
  
  const gate = document.getElementById(
    'email-gate'
  );

  if (hasAccess) {
    // Already signed up — hide gate immediately
    if (gate) gate.style.display = 'none';
    return;
  }

  // Load and display subscriber count
  async function loadCounter() {
    try {
      const res = await fetch('/api/subscriber-count');
      const data = await res.json();
      const count = data.count || 0;
      const pct = Math.min(
        (count / 10000) * 100, 100
      ).toFixed(1);

      const counterEl = document.getElementById(
        'user-counter'
      );
      const barEl = document.getElementById(
        'counter-bar'
      );

      if (counterEl) {
        // Animate count up
        let start = 0;
        const end = count;
        const duration = 1000;
        const step = end / (duration / 16);
        
        const timer = setInterval(() => {
          start += step;
          if (start >= end) {
            start = end;
            clearInterval(timer);
          }
          counterEl.textContent = (
            Math.floor(start).toLocaleString()
          );
        }, 16);
      }

      if (barEl) {
        setTimeout(() => {
          barEl.style.width = pct + '%';
        }, 100);
      }

    } catch(e) {
      const counterEl = document.getElementById(
        'user-counter'
      );
      if (counterEl) counterEl.textContent = '0';
    }
  }

  loadCounter();

  // Handle form submission
  const form = document.getElementById(
    'gate-form'
  );
  const errorEl = document.getElementById(
    'gate-error'
  );
  const submitBtn = document.getElementById(
    'gate-submit'
  );

  if (form) {
    form.addEventListener('submit', async (e) => {
      e.preventDefault();

      const email = document.getElementById(
        'gate-email'
      )?.value?.trim();
      const name = document.getElementById(
        'gate-name'
      )?.value?.trim();

      // Basic validation
      if (!email || !email.includes('@')) {
        errorEl.textContent = (
          'Please enter a valid email address.'
        );
        errorEl.classList.remove('hidden');
        return;
      }

      errorEl.classList.add('hidden');
      submitBtn.textContent = 'Getting your access...';
      submitBtn.disabled = true;

      try {
        const res = await fetch('/api/gate-signup', {
          method: 'POST',
          headers: {
            'Content-Type': 'application/json'
          },
          body: JSON.stringify({ email, name })
        });

        const data = await res.json();

        if (data.success) {
          // Grant access
          localStorage.setItem('mora_access', '1');
          localStorage.setItem('mora_email', email);

          // Animate gate out
          gate.style.transition = (
            'opacity 0.4s ease'
          );
          gate.style.opacity = '0';
          setTimeout(() => {
            gate.style.display = 'none';
          }, 400);

        } else {
          // Still grant access even if 
          // server had an issue
          localStorage.setItem('mora_access', '1');
          gate.style.display = 'none';
        }

      } catch(err) {
        // Network error — still grant access
        // Never block user for server issues
        localStorage.setItem('mora_access', '1');
        gate.style.display = 'none';
      }
    });
  }

})();
</script>


=== TASK 3: BACKEND ROUTES IN app.py ===

Add these two routes:

import csv
from datetime import datetime

EMAIL_LIST_FILE = 'email_subscribers.csv'

@app.route('/api/gate-signup', methods=['POST'])
def gate_signup():
    """
    Save email from gate form.
    Always returns success so gate 
    never blocks a user.
    """
    try:
        data = request.json or {}
        email = data.get('email', '').strip().lower()
        name  = data.get('name', '').strip()

        if not email or '@' not in email:
            return jsonify({
                "success": False,
                "error": "Invalid email"
            }), 400

        # Check if already exists
        existing = _load_emails()
        if email in existing:
            logger.info(
                f"[GATE] Returning user: {email}"
            )
            return jsonify({
                "success": True,
                "existing": True
            })

        # Save to CSV
        _save_email(email, name)

        count = len(_load_emails())
        logger.info(
            f"[GATE] New signup: {email} "
            f"(total: {count})"
        )

        return jsonify({
            "success": True,
            "count":   count
        })

    except Exception as e:
        logger.error(f"[GATE] Signup error: {e}")
        # Always return success to never 
        # block a user from the dashboard
        return jsonify({"success": True})


@app.route('/api/subscriber-count')
def subscriber_count():
    """Return current subscriber count."""
    try:
        count = len(_load_emails())
        return jsonify({
            "count": count,
            "spots_remaining": max(
                0, 10000 - count
            )
        })
    except Exception:
        return jsonify({"count": 0})


@app.route('/api/subscribers/export')
def export_subscribers():
    """
    Export full email list as CSV.
    Add a secret key check before using 
    in production.
    """
    try:
        from flask import send_file
        if not os.path.exists(EMAIL_LIST_FILE):
            return jsonify({"error": "No list yet"})
        return send_file(
            EMAIL_LIST_FILE,
            mimetype='text/csv',
            as_attachment=True,
            download_name='mora_bets_emails.csv'
        )
    except Exception as e:
        return jsonify({"error": str(e)}), 500


def _load_emails():
    """Load all emails from CSV file."""
    emails = set()
    try:
        if os.path.exists(EMAIL_LIST_FILE):
            with open(
                EMAIL_LIST_FILE, 'r'
            ) as f:
                reader = csv.DictReader(f)
                for row in reader:
                    if row.get('email'):
                        emails.add(
                            row['email'].lower()
                        )
    except Exception as e:
        logger.error(
            f"[GATE] Load emails error: {e}"
        )
    return emails


def _save_email(email: str, name: str = ''):
    """Append new email to CSV file."""
    try:
        file_exists = os.path.exists(
            EMAIL_LIST_FILE
        )
        with open(
            EMAIL_LIST_FILE, 'a', newline=''
        ) as f:
            writer = csv.DictWriter(
                f,
                fieldnames=[
                    'email', 'name', 
                    'signed_up_at'
                ]
            )
            if not file_exists:
                writer.writeheader()
            writer.writerow({
                'email':         email,
                'name':          name,
                'signed_up_at':  datetime.utcnow(
                ).isoformat()
            })
    except Exception as e:
        logger.error(
            f"[GATE] Save email error: {e}"
        )
