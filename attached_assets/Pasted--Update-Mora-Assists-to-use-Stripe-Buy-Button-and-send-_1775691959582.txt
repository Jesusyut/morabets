    updates subscriber status.

    import csv
    import tempfile

        'STRIPE_SECRET_KEY'

        return """
        <html><body style="font-family:Inter;
        text-align:center;padding:60px;">
        <h2>Link invalid.</h2>
        <p>Please contact us at 
        picks@morabets.com</p>
        </body></html>
        """, 400

    try:
        # Find subscription ID for this email
        FILE = 'mora_assists_subscribers.csv'
        sub_id = None
        name   = ''

        if os.path.exists(FILE):
            with open(FILE) as f:
                reader = csv.DictReader(f)
                for row in reader:
                    if row.get(
                        'email', ''
                    ).lower() == email.lower():
                        sub_id = row.get(
                            'stripe_subscription_id'
                        )
                        name = row.get('name', '')

        if sub_id:

                stripe.Subscription.cancel(sub_id)

                logger.warning(
