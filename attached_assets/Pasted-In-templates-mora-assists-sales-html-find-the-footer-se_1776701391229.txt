<script>
async function submitFridaySales() {
  const email = document.getElementById(
    'friday-email-sales'
  ).value.trim();
  if (!email || !email.includes('@')) return;

  try {
    const res = await fetch(
      'https://hooks.zapier.com/hooks/catch/22797974/uj2zhly/',
      {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json'
        },
        body: JSON.stringify({
          email: email,
          source: 'friday_capture_sales_page'
        })
      }
    );
    document.getElementById(
      'friday-signup-sales'
    ).style.display = 'none';
    document.getElementById(
      'friday-success-sales'
    ).style.display = 'block';
    localStorage.setItem(
      'mora_friday_captured', 'true'
    );
  } catch(e) {
    console.log('Submit error:', e);
  }
}
</script>
