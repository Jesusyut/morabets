const barMessages = [

  "FanDuel vs DraftKings on the same pick can "+

];

let barMsgIndex = 0;
const barText = document.getElementById(
  'bar-rotating-text'
);

setInterval(() => {
  barMsgIndex = (barMsgIndex + 1) % barMessages.length;
  barText.style.opacity = '0';
  setTimeout(() => {
    barText.textContent = barMessages[barMsgIndex];
    barText.style.opacity = '1';
  }, 300);
  barText.style.transition = 'opacity 0.3s';
}, 5000);

// Dismiss button
document.getElementById('bar-dismiss')
  .addEventListener('click', () => {
    document.getElementById('sticky-bar')
      .style.transform = 'translateY(100%)';
    // Remember for session
    sessionStorage.setItem('barDismissed', '1');
  });

// Don't show if dismissed this session
if (sessionStorage.getItem('barDismissed')) {
  document.getElementById('sticky-bar')
    .style.transform = 'translateY(100%)';
}
</script>

Add a simple email list to the app.

In app.py add:

EMAIL_LIST_FILE = 'email_subscribers.json'

def save_subscriber(email: str) -> bool:
    """Save email to subscriber list."""
    try:
        try:
            with open(EMAIL_LIST_FILE, 'r') as f:
                subscribers = json.load(f)
        except FileNotFoundError:
            subscribers = []
        
        # Deduplicate
        if email not in subscribers:
            subscribers.append(email)
            with open(EMAIL_LIST_FILE, 'w') as f:
                json.dump(subscribers, f, indent=2)
            logger.info(
                f'[EMAIL LIST] Added: {email} '
                f'({len(subscribers)} total)'
            )
        return True
    except Exception as e:
        logger.error(f'[EMAIL LIST] Error: {e}')
        return False

@app.route('/api/subscribe', methods=['POST'])
def subscribe():
    data = request.json or {}
    email = data.get('email', '').strip().lower()
    
    if not email or '@' not in email:
        return jsonify({"error": "Invalid email"}), 400
    
    success = save_subscriber(email)
    return jsonify({
        "success": success,
        "message": "You're in. Picks hit your inbox 
                    before first pitch."
    })
