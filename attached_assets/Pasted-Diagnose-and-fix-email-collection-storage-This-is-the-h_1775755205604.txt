  import csv, os
  from datetime import datetime

  OLD = 'email_subscribers.csv'
  NEW = '/var/data/email_subscribers.csv'
  FIELDS = ['email','name','signed_up_at']

  existing = set()
  rows = []

  # Load new file first
  if os.path.exists(NEW):
      with open(NEW) as f:
          for r in csv.DictReader(f):
              if r.get('email'):
                  existing.add(r['email'].lower())
                  rows.append(r)

  # Load old file and merge
  if os.path.exists(OLD):
      with open(OLD) as f:
          for r in csv.DictReader(f):
              email = r.get('email','').lower()
              if email and email not in existing:
                  existing.add(email)
                  rows.append(r)
                  print(f'Recovered: {email}')

  # Write merged file
  os.makedirs('/var/data', exist_ok=True)
  with open(NEW, 'w', newline='') as f:
      writer = csv.DictWriter(f, fieldnames=FIELDS)
      writer.writeheader()
      writer.writerows(rows)

  print(f'Total emails saved: {len(rows)}')
  "

=== STEP 5: ADD BACKUP WRITE ===

In app.py find _save_email() function.

After the existing write to EMAIL_LIST_FILE
add a second backup write to ensure 
double redundancy:

def _save_email(email: str, name: str = ''):
    try:
        file_exists = os.path.exists(
            EMAIL_LIST_FILE
        )
        with open(
            EMAIL_LIST_FILE, 'a', newline=''
        ) as f:
            writer = csv.DictWriter(
                f,
                fieldnames=[
                    'email', 'name',
                    'signed_up_at'
                ]
            )
            if not file_exists:
                writer.writeheader()
            writer.writerow({
                'email':        email,
                'name':         name,
                'signed_up_at': datetime.utcnow(
                ).isoformat()
            })
        
        # ── BACKUP WRITE ──────────────────
        # Secondary log in case primary fails
        backup = '/var/data/emails_backup.txt'
        with open(backup, 'a') as b:
            b.write(
                f"{datetime.utcnow().isoformat()}"
                f"|{email}|{name}\n"
            )
        # ── END BACKUP ────────────────────

        logger.info(
            f"[GATE] Saved: {email} "
            f"→ {EMAIL_LIST_FILE}"
        )

    except Exception as e:
        logger.error(
            f"[GATE] Save failed: {e}"
        )
        # Emergency fallback — log to 
        # application logs at minimum
        # so email is never silently lost
        logger.error(
            f"[GATE] LOST EMAIL: {email}"
        )
