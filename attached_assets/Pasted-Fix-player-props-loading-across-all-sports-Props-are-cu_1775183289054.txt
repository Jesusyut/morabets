=== ROOT CAUSE CONFIRMED ===

Two bugs broke props after recent updates:

Bug 1 — VALID_BOOKS case mismatch (odds_api.py)
  VALID_BOOKS = {"DraftKings", "FanDuel", "BetMGM"}
  But API returns "draftkings", "fanduel", "betmgm"
  lowercase from the bookmakers key field.
  The title field returns "DraftKings" etc but the
  filter was checking the wrong field inconsistently.
  Result: all books silently filtered out = 0 props.

Bug 2 — EV engine requirement on props
  After the EV upgrade, group_props_by_player
  calls evaluate_pick() from ev_engine.py which
  requires a minimum number of books with BOTH
  over AND under prices. Props often have 1-2 books
  with only the over posted. EV engine rejects them.
  Result: valid props filtered before display.

        &bookmakers=draftkings,fanduel,betmgm,...

    {
      "name": "Over",            <- side
      "description": "Judge",    <- player name

    }
    
    Player name MUST come from outcome["description"]
    Side MUST come from outcome["name"]

    future = now + timedelta(hours=24)
    start  = now.strftime('%Y-%m-%dT%H:%M:%SZ')

                            player_map.items()
                        ):
                            if sides[
                                "over_price"
                            ] is None:
                                continue

                            all_props.append({
                                "player":      pname,

                )

    # Stat breakdown log

    for p in all_props:
        k = p.get("stat", "?")
        stat_counts[k] = stat_counts.get(k, 0) + 1

    print(
        f"[PROPS] {len(all_props)} raw props "
        f"from {len(events)} events: {stat_counts}"

    return all_props

def group_props_by_player(props):
    """
    Group raw props by player+stat+line.
    Calculate no-vig when both sides available.
    Calculate implied when over only available.
    Filter: min 52% no-vig, min 58% implied.
    Sort: highest probability first.
    NO EV ENGINE INVOLVEMENT.
    """
    from probability import (
        no_vig_probability,
        get_confidence_tier,
        implied_probability_single,
        get_confidence_tier_implied
    )

    # Try to import stat display helper
    try:
        from prop_deduplication import (
            get_stat_display_name
        )
    except Exception:
        def get_stat_display_name(stat):
            return stat.replace("_", " ").title()

    # Group all rows by player+stat+line
    groups = {}

    for p in props:
        player = p.get("player", "")
        stat   = p.get("stat", "")
        line   = p.get("line")

        if not player or not stat:
            continue

        key = f"{player}|{stat}|{line}"

        if key not in groups:
            groups[key] = {
                "player":    player,
                "stat":      stat,
                "stat_label": get_stat_display_name(
                    stat
                ),
                "line":      line,
                "home_team": p.get("home_team", ""),
                "away_team": p.get("away_team", ""),
                "matchup":   p.get("matchup", ""),
                "game_time": p.get("game_time", ""),
                "sport":     p.get("sport", "MLB"),
                "all_books": []
            }

        # Accept both paired and legacy shapes
        over_p  = (
            p.get("over_price") or p.get("odds")
        )
        under_p = p.get("under_price")

        if over_p is not None:
            groups[key]["all_books"].append({
                "book":        p.get("bookmaker",""),
                "over_price":  over_p,
                "under_price": under_p
            })

    result = []

    for key, group in groups.items():
        books = group["all_books"]
        if not books:
            continue

        # Separate two-sided from one-sided
        two_sided = [
            b for b in books
            if b.get("over_price") is not None
            and b.get("under_price") is not None
        ]
        one_sided = [
            b for b in books
            if b.get("over_price") is not None
            and b.get("under_price") is None
        ]

        if two_sided:
            # Full no-vig — use sharpest book
            # (lowest combined implied = least vig)
            def vig_sum(b):
                def imp(o):
                    return (
                        100/(o+100) if o > 0
                        else abs(o)/(abs(o)+100)
                    )
                return (
                    imp(b["over_price"]) +
                    imp(b["under_price"])
                )

            sharpest = min(two_sided, key=vig_sum)
            best     = max(
                two_sided,
                key=lambda b: b["over_price"]
            )

            nv = no_vig_probability(
                sharpest["over_price"],
                sharpest["under_price"]
            )

            # Hard minimum: 52% no-vig
            if nv < 52.0:
                continue

            tier = get_confidence_tier(nv)

            result.append({
                **group,
                "all_books":       two_sided,
                "no_vig_prob":     nv,
                "prob_label":      "True Prob",
                "prob_type":       "no_vig",
                "confidence_tier": tier,
                "best_book":       best["book"],
                "best_over_price": best["over_price"],
                "implied_only":    False,
                "passes_threshold": False,
                "ev_pct":          None,
                "surfaced":        True,
                "enriched":        True
            })

        elif one_sided:
            # Implied only — single side
            best    = max(
                one_sided,
                key=lambda b: b["over_price"]
            )
            over_p  = best["over_price"]
            implied = implied_probability_single(
                over_p
            )

            # Hard minimum: 58% implied
            # (accounts for vig in single price)
            if implied < 58.0:
                continue

            tier = get_confidence_tier_implied(implied)

            result.append({
                **group,
                "all_books":       one_sided,
                "no_vig_prob":     implied,
                "prob_label":      "Implied",
                "prob_type":       "implied",
                "confidence_tier": tier,
                "best_book":       best["book"],
                "best_over_price": over_p,
                "implied_only":    True,
                "passes_threshold": False,
                "ev_pct":          None,
                "surfaced":        True,
                "enriched":        True
            })

        else:
            continue

    # Sort: highest probability first
    result.sort(
        key=lambda p: -p.get("no_vig_prob", 0)
    )

    print(
        f"[PROPS] {len(props)} raw → "
        f"{len(result)} after grouping/filtering"
    )
    return result

@app.route("/api/refresh-props", methods=["POST"])
def refresh_props():
    """
    Manual props refresh endpoint.
    Called by the dashboard refresh button.
    Fetches fresh props from Odds API right now.
    Burns API quota — protect with rate limiting.
    """
    # Simple rate limit: max once per 30 minutes
    last_refresh = _cache_get(
        "last_manual_refresh", ttl_seconds=1800
    )
    if last_refresh:
        return jsonify({
            "status":  "rate_limited",
            "message": "Props refreshed less than "
                       "30 minutes ago. "
                       "Using cached data."
        }), 429

    try:
        from threading import Thread
        
        def refresh_all():
            _fetch_and_process_mlb_props()
            _fetch_and_process_nhl_props()

        t = Thread(target=refresh_all, daemon=True)
        t.start()

        _cache_set("last_manual_refresh", True)

        return jsonify({
            "status":  "refreshing",
            "message": "Props updating in background. "
                       "Reload in 30 seconds."
        })

    except Exception as e:
        return jsonify({"error": str(e)}), 500
