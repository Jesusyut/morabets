    message = client.messages.create(
        model='claude-sonnet-4-5-20250929',
        max_tokens=1000,
        system=SYSTEM_PROMPT,

            {
                'role': 'user',
                'content': data.get(

                )
            }
        ],
        timeout=45.0
    )
    raw = message.content[0].text.strip()

except anthropic.APITimeoutError:

    )
    return jsonify({
        'response': 'Analysis failed. Try again.'
    }), 500
