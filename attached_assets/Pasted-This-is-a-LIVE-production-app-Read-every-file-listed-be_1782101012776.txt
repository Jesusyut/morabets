═══════════════════════════════════════
WHAT THIS FEATURE IS
═══════════════════════════════════════

A Context Edge Analyst panel that
appears when a user clicks a specific
button on any prop card or game line
card. It sends the line data to the
Anthropic Claude API and returns a
structured JSON verdict displayed
inside the dashboard — no page reload,
no navigation, instant inline result.

This is NOT a picks generator.
It analyzes whether a specific line
has a context edge beyond what the
no-vig math alone shows.

@app.route('/api/context-edge', methods=['POST'])
def context_edge_analysis():
    """
    Receives a betting line from the
    frontend and returns Claude's
    Context Edge analysis as JSON.
    """
    import anthropic
    import json

    try:
        data = request.get_json()
        if not data:
            return jsonify(
                {'error': 'No data provided'}
            ), 400

        # Build the user message from
        # whatever context the card has
        player = data.get('player', '')
        stat = data.get('stat', '')
        line = data.get('line', '')
        odds = data.get('best_over_price', '')
        no_vig = data.get('no_vig_prob', '')
        matchup = data.get('matchup', '')
        sport = data.get('sport', '')
        environment = data.get(
            'environment', 'No label'
        )
        home_team = data.get('home_team', '')
        away_team = data.get('away_team', '')
        all_books = data.get('all_books', [])
        market_type = data.get(
            'market_type', 'player_prop'
        )

        # Calculate sportsbook implied prob
        # from raw odds for the message
        if odds and odds != 0:
            if odds < 0:
                impl = abs(odds) / (
                    abs(odds) + 100
                ) * 100
            else:
                impl = 100 / (odds + 100) * 100
            implied_prob = f"{impl:.1f}%"
        else:
            implied_prob = "Unknown"

        # Format books for context
        books_str = ', '.join([
            f"{b.get('book','')} "
            f"({b.get('over_price','')})"
            for b in all_books[:5]
        ]) if all_books else 'Not provided'

        user_message = f"""
Analyze this betting line for context edge:

SPORT: {sport}
MATCHUP: {matchup}
HOME TEAM: {home_team}
AWAY TEAM: {away_team}
ENVIRONMENT: {environment}
MARKET TYPE: {market_type}
PLAYER: {player}
STAT: {stat} OVER {line}
BEST ODDS: {odds} (American)
SPORTSBOOK IMPLIED PROBABILITY: {implied_prob}
NO-VIG TRUE PROBABILITY: {no_vig}%
BOOKS CARRYING THIS LINE: {books_str}

Additional context provided by user:
{data.get('user_context', 'None provided')}

Analyze the context edge for this line.
Return ONLY valid JSON matching the
required schema exactly.
Do not invent data not provided above.
List any missing data under missing_data.
"""

        SYSTEM_PROMPT = """You are Mora Bets Context Edge Analyst.
Your job is to decide whether selected betting lines have a CONTEXT EDGE.
A context edge means matchup-specific information may make the true probability better than the no-vig market probability or better than the sportsbook implied probability.

You must analyze:
- Price value
- No-vig probability vs sportsbook implied probability
- Starting pitcher/starter/goalie context when relevant
- Injuries and lineup context
- Weather/venue/park context
- Rest/travel/fatigue
- Recent form
- Matchup style
- Splits
- Alt-line safety
- Market movement if provided
- Missing data

You must not invent data. If data is not provided, list it under missing_data and reduce confidence accordingly.

You must return ONLY valid JSON. No preamble. No explanation outside the JSON.

Required JSON schema:
{
  "verdict": "PLAY | SMALL PLAY | PASS",
  "confidence": 1-10,
  "risk_level": "LOW | MEDIUM | HIGH",
  "context_edge_score": 1-10,
  "price_value_score": 1-10,
  "no_vig_probability": "",
  "book_odds": "",
  "sportsbook_implied_probability": "",
  "estimated_true_probability_range": "",
  "estimated_edge": "",
  "context_supporting_bet": [],
  "context_against_bet": [],
  "missing_data": [],
  "best_line": "",
  "alt_line_suggestion": "",
  "suggested_unit_size": "",
  "reason": "",
  "what_would_make_this_a_pass": ""
}

Decision rules — strictly enforced:
- If sportsbook implied probability exceeds no-vig probability and context is weak: verdict MUST be PASS
- If odds are heavy favorite pricing and true probability does not clearly exceed break-even: verdict MUST be PASS
- If plus-money or fair-priced AND context strongly supports: consider PLAY or SMALL PLAY
- If important context data is missing: confidence cannot exceed 7
- If both price_value_score AND context_edge_score are below 7: verdict MUST be PASS
- Prefer disciplined passes over forced plays
- Never recommend a bet solely because no-vig probability is high"""

        client = anthropic.Anthropic(
            api_key=os.environ.get(
                'ANTHROPIC_API_KEY'
            )
        )

        message = client.messages.create(
            model='claude-sonnet-4-6',
            max_tokens=1500,
            system=SYSTEM_PROMPT,
            messages=[
                {
                    'role': 'user',
                    'content': user_message
                }
            ]
        )

        raw = message.content[0].text.strip()

        # Strip markdown fences if present
        if raw.startswith('```'):
            raw = raw.split('```')[1]
            if raw.startswith('json'):
                raw = raw[4:]
        raw = raw.strip()

        result = json.loads(raw)
        return jsonify(result)

    except json.JSONDecodeError as e:
        return jsonify({
            'error': 'Invalid JSON from model',
            'raw': raw if 'raw' in locals()
                   else 'No response'
        }), 500
    except Exception as e:
        logger.error(
            f'Context edge error: {e}'
        )
        return jsonify(
            {'error': str(e)}
        ), 500

// Stores current prop being analyzed
// for re-analyze with context
let _currentContextProp = null;

function openContextEdgePanel(prop) {
  _currentContextProp = prop;
  const overlay = document.getElementById(
    'context-edge-overlay'
  );
  overlay.style.display = 'flex';
  document.body.style.overflow = 'hidden';

  // Set header
  const title = prop.player
    ? `${prop.player} — ${prop.stat} OVER ${prop.line}`
    : `${prop.home_team} vs ${prop.away_team}`;
  document.getElementById(
    'ce-title'
  ).textContent = title;
  document.getElementById(
    'ce-subtitle'
  ).textContent =
    `${prop.matchup} · ${prop.sport}`;

  // Show loading, hide result + error
  document.getElementById(
    'ce-loading'
  ).style.display = 'block';
  document.getElementById(
    'ce-result'
  ).style.display = 'none';
  document.getElementById(
    'ce-error'
  ).style.display = 'none';
  document.getElementById(
    'ce-user-context'
  ).value = '';

  fetchContextEdge(prop, '');
}

async function fetchContextEdge(
  prop, userContext
) {
  try {
    const payload = {
      ...prop,
      user_context: userContext
    };

    const response = await fetch(
      '/api/context-edge',
      {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json'
        },
        body: JSON.stringify(payload)
      }
    );

    if (!response.ok) {
      throw new Error(
        `HTTP ${response.status}`
      );
    }

    const data = await response.json();

    if (data.error) {
      throw new Error(data.error);
    }

    renderContextEdgeResult(data);

  } catch(err) {
    console.error('Context Edge error:', err);
    document.getElementById(
      'ce-loading'
    ).style.display = 'none';
    document.getElementById(
      'ce-error'
    ).style.display = 'block';
  }
}

function renderContextEdgeResult(d) {
  document.getElementById(
    'ce-loading'
  ).style.display = 'none';
  document.getElementById(
    'ce-result'
  ).style.display = 'block';

  // Verdict color
  const verdictEl = document.getElementById(
    'ce-verdict'
  );
  verdictEl.textContent = d.verdict || '-';
  verdictEl.style.color =
    d.verdict === 'PLAY' ? '#4CBB17' :
    d.verdict === 'SMALL PLAY' ? '#EAB308' :
    '#EF4444';

  // Scores
  document.getElementById(
    'ce-confidence'
  ).textContent = `${d.confidence || '-'}/10`;
  document.getElementById(
    'ce-context-score'
  ).textContent =
    `${d.context_edge_score || '-'}/10`;
  document.getElementById(
    'ce-price-score'
  ).textContent =
    `${d.price_value_score || '-'}/10`;
  document.getElementById(
    'ce-risk'
  ).textContent =
    `Risk: ${d.risk_level || '-'}`;

  // Probabilities
  document.getElementById(
    'ce-novigprob'
  ).textContent =
    d.no_vig_probability || '-';
  document.getElementById(
    'ce-impliedprob'
  ).textContent =
    d.sportsbook_implied_probability || '-';

  const edgeEl = document.getElementById(
    'ce-edge'
  );
  edgeEl.textContent =
    d.estimated_edge || '-';
  const edge = parseFloat(
    (d.estimated_edge || '0').replace('%','')
  );
  edgeEl.style.color = edge > 0
    ? '#4CBB17' : '#EF4444';

  // Reason
  document.getElementById(
    'ce-reason'
  ).textContent = d.reason || '-';

  // Context for
  const forEl = document.getElementById(
    'ce-for'
  );
  forEl.innerHTML = (
    d.context_supporting_bet || []
  ).map(
    c => `<li style="margin-bottom:4px;">${c}</li>`
  ).join('') || '<li>None identified</li>';

  // Context against
  const againstEl = document.getElementById(
    'ce-against'
  );
  againstEl.innerHTML = (
    d.context_against_bet || []
  ).map(
    c => `<li style="margin-bottom:4px;">${c}</li>`
  ).join('') || '<li>None identified</li>';

  // Missing data
  const missing = d.missing_data || [];
  const missingWrap = document.getElementById(
    'ce-missing-wrap'
  );
  if (missing.length > 0) {
    missingWrap.style.display = 'block';
    document.getElementById(
      'ce-missing'
    ).innerHTML = missing.map(
      m => `<li style="margin-bottom:4px;">${m}</li>`
    ).join('');
  } else {
    missingWrap.style.display = 'none';
  }

  // Bet specs
  document.getElementById(
    'ce-bestline'
  ).textContent = d.best_line || '-';
  document.getElementById(
    'ce-units'
  ).textContent =
    d.suggested_unit_size || '-';

  // Pass rule
  document.getElementById(
    'ce-passrule'
  ).textContent =
    d.what_would_make_this_a_pass || '-';
}

function reAnalyzeWithContext() {
  if (!_currentContextProp) return;
  const ctx = document.getElementById(
    'ce-user-context'
  ).value.trim();

  document.getElementById(
    'ce-result'
  ).style.display = 'none';
  document.getElementById(
    'ce-loading'
  ).style.display = 'block';

  fetchContextEdge(_currentContextProp, ctx);
}

function closeContextEdge() {
  document.getElementById(
    'context-edge-overlay'
  ).style.display = 'none';
  document.body.style.overflow = '';
  _currentContextProp = null;
}

// Close on overlay background click
document.getElementById(
  'context-edge-overlay'
).addEventListener('click', function(e) {
  if (e.target === this) closeContextEdge();
});

// Close on Escape key
document.addEventListener(
  'keydown', function(e) {
    if (e.key === 'Escape') {
      closeContextEdge();
    }
  }
);
