let _ceChatOpen = false;
let _ceAuthed = false;

function toggleContextChat() {
  const panel = document.getElementById(
    'ce-chat-panel'
  );
  const tooltip = document.getElementById(
    'ce-tooltip'
  );

  // Hide tooltip permanently after
  // first click
  if (tooltip) tooltip.style.display = 'none';

  _ceChatOpen = !_ceChatOpen;

  if (_ceChatOpen) {
    panel.style.display = 'flex';
    panel.style.flexDirection = 'column';

    // Show passcode gate or input area
    // depending on auth state
    document.getElementById(
      _ceAuthed
        ? 'ce-input-area'
        : 'ce-passcode-gate'
    ).style.display = 'block';

    if (_ceAuthed) {
      document.getElementById(
        'ce-passcode-gate'
      ).style.display = 'none';
      setTimeout(() => {
        document.getElementById(
          'ce-input'
        ).focus();
      }, 100);
    }
  } else {
    panel.style.display = 'none';
  }
}

async function submitPasscode() {
  const input = document.getElementById(
    'ce-passcode-input'
  );
  const error = document.getElementById(
    'ce-passcode-error'
  );
  const code = input.value.trim();

  if (!code) return;

  // Validate against backend
  try {
    const res = await fetch(
      '/api/context-edge',
      {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json'
        },
        body: JSON.stringify({
          passcode: code,
          validate_only: true,
          user_message: 'ping'
        })
      }
    );

    if (res.status === 403) {
      error.style.display = 'block';
      input.value = '';
      sessionStorage.removeItem(
        'ce_passcode'
      );
      return;
    }

    // Auth success
    sessionStorage.setItem('ce_passcode', code);
    _ceAuthed = true;
    error.style.display = 'none';
    document.getElementById(
      'ce-passcode-gate'
    ).style.display = 'none';
    document.getElementById(
      'ce-input-area'
    ).style.display = 'block';
    setTimeout(() => {
      document.getElementById('ce-input')
        .focus();
    }, 100);

  } catch(e) {
    error.textContent = 'Connection error.';
    error.style.display = 'block';
  }
}
