def load_full_board():
    """
    Load all cached props and game lines
    across every sport.
    As new sports are added they will be
    picked up automatically here.
    """
    board = {
        "props": [],
        "lines": [],
        "sports_found": []
    }

    # MLB props
    mlb_props = load_props_from_file(
        'mlb_props_cache.json'
    )
    if mlb_props:
        board["props"].extend(mlb_props)
        board["sports_found"].append("MLB")

    # NHL props
    nhl_props = load_props_from_file(
        'nhl_props_cache.json'
    )
    if nhl_props:
        board["props"].extend(nhl_props)
        if "NHL" not in board["sports_found"]:
            board["sports_found"].append("NHL")

    # NFL props (when in season)
    try:
        nfl_props = load_props_from_file(
            'nfl_props_cache.json'
        )
        if nfl_props:
            board["props"].extend(nfl_props)
            board["sports_found"].append("NFL")
    except Exception:
        pass

    # Game lines — read from any lines cache
    for cache_file in [
        'mlb_lines_cache.json',
        'nhl_lines_cache.json',
        'nfl_lines_cache.json'
    ]:
        try:
            lines = load_props_from_file(cache_file)
            if lines:
                board["lines"].extend(lines)
        except Exception:
            pass

    # Filter props to quality threshold
    board["props"] = [
        p for p in board["props"]
        if p.get("no_vig_prob", 0) >= 55
    ]

    # Filter lines to quality threshold
    board["lines"] = [
        l for l in board["lines"]
        if l.get("no_vig_prob", 0) >= 60
    ]

    logger.info(
        f"[ASSISTS] Board loaded: "
        f"{len(board['props'])} props, "
        f"{len(board['lines'])} lines, "
        f"sports: {board['sports_found']}"
    )
    return board

def select_picks_with_llm(board):
    """
    Send board data to Claude and get
    5 curated picks back as JSON.
    """
    if not ANTHROPIC_API_KEY:
        logger.error(
            "[ASSISTS] No ANTHROPIC_API_KEY"
        )
        return None

    client = anthropic.Anthropic(
        api_key=ANTHROPIC_API_KEY
    )

    # Prepare board summary for LLM
    board_summary = {
        "props":             board["props"][:100],
        "lines":             board["lines"][:50],
        "sports_available":  board["sports_found"],
        "total_props":       len(board["props"]),
        "total_lines":       len(board["lines"]),
        "generated_at":      datetime.utcnow(
        ).isoformat()
    }

    user_message = (
        f"Today's board data:\n"
        f"{json.dumps(board_summary, indent=2)}"
    )

    try:
        response = client.messages.create(
            model="claude-sonnet-4-20250514",
            max_tokens=2000,
            system=SELECTION_PROMPT,
            messages=[
                {
                    "role":    "user",
                    "content": user_message
                }
            ]
        )

        raw = response.content[0].text.strip()

        # Clean any markdown if present
        raw = raw.replace(
            "```json", ""
        ).replace("```", "").strip()

        picks_data = json.loads(raw)

        logger.info(
            f"[ASSISTS] LLM selected "
            f"{len(picks_data.get('picks',[]))} "
            f"picks"
        )
        return picks_data

    except json.JSONDecodeError as e:
        logger.error(
            f"[ASSISTS] JSON parse error: {e}"
            f"\nRaw: {raw[:500]}"
        )
        return None
    except Exception as e:
        logger.error(
            f"[ASSISTS] LLM error: {e}"
        )
        return None

def load_active_subscribers():
    """
    Load all active Mora Assists subscribers.
    Status must be 'active' or 'trial'.
    """
    import csv
    subscribers = []
    try:
        if os.path.exists(
            'mora_assists_subscribers.csv'
        ):
            with open(
                'mora_assists_subscribers.csv'
            ) as f:
                reader = csv.DictReader(f)
                for row in reader:
                    if row.get('status') in [
                        'active', 'trial'
                    ]:
                        subscribers.append(row)
    except Exception as e:
        logger.error(
            f"[ASSISTS] Load subscribers: {e}"
        )
    return subscribers

def send_picks_email(
    to_email, subject, html_content
):
    """Send one email via SendGrid."""
    try:
        sg = sendgrid.SendGridAPIClient(
            api_key=SENDGRID_API_KEY
        )
        message = Mail(
            from_email=FROM_EMAIL,
            to_emails=to_email,
            subject=subject,
            html_content=html_content
        )
        response = sg.send(message)
        return response.status_code in [
            200, 201, 202
        ]
    except Exception as e:
        logger.error(
            f"[ASSISTS] Send failed "
            f"{to_email}: {e}"
        )
        return False

def run_daily_assists():
    """
    Main job called by scheduler at 10:30 AM ET.
    Loads board, selects picks, sends emails.
    """
    logger.info(
        "[ASSISTS] Starting daily run..."
    )

    # Load full board across all sports
    board = load_full_board()

    if not board["props"] and not board["lines"]:
        logger.warning(
            "[ASSISTS] Board empty — "
            "no picks to send today"
        )
        return

    # Select picks with LLM
    picks_data = select_picks_with_llm(board)

    if not picks_data:
        logger.error(
            "[ASSISTS] LLM returned no picks"
        )
        return

    # Save today's picks to file
    # (for debugging and record keeping)
    today_str = datetime.now().strftime(
        "%Y-%m-%d"
    )
    with open(
        f'assists_picks_{today_str}.json', 'w'
    ) as f:
        json.dump(picks_data, f, indent=2)

    logger.info(
        f"[ASSISTS] Picks saved to "
        f"assists_picks_{today_str}.json"
    )

    # Format email
    subject, html = format_picks_email(
        picks_data
    )
    if not subject or not html:
        logger.error(
            "[ASSISTS] Email format failed"
        )
        return

    # Load active subscribers
    subscribers = load_active_subscribers()
    logger.info(
        f"[ASSISTS] Sending to "
        f"{len(subscribers)} subscribers"
    )

    if not subscribers:
        logger.warning(
            "[ASSISTS] No active subscribers yet"
        )
        return

    # Send to all active subscribers
    sent = 0
    failed = 0
    for sub in subscribers:
        email = sub.get('email')
        if not email:
            continue
        success = send_picks_email(
            email, subject, html
        )
        if success:
            sent += 1
        else:
            failed += 1

    logger.info(
        f"[ASSISTS] Sent: {sent} "
        f"Failed: {failed}"
    )
    return {
        "sent":   sent,
        "failed": failed,
        "picks":  len(
            picks_data.get('picks', [])
        )
    }

def stripe_webhook():
    """
    Handle Stripe subscription events.
    Updates subscriber status automatically.
    """
    payload = request.data
    sig     = request.headers.get(
        'Stripe-Signature'
    )
    secret  = os.environ.get(
        'STRIPE_WEBHOOK_SECRET'
    )

    try:
        event = stripe.Webhook.construct_event(
            payload, sig, secret
        )
    except Exception as e:
        logger.error(
            f"[STRIPE] Webhook error: {e}"
        )
        return jsonify({'error': str(e)}), 400

    _handle_stripe_event(event)
    return jsonify({'received': True})


def _handle_stripe_event(event):
    """Process Stripe subscription events."""
    import csv
    from datetime import datetime, timedelta

    event_type = event['type']
    data       = event['data']['object']

    logger.info(
        f"[STRIPE] Event: {event_type}"
    )

    if event_type == (
        'customer.subscription.created'
    ):
        # New subscriber — add to CSV
        customer = stripe.Customer.retrieve(
            data['customer']
        )
        email    = customer.get('email', '')
        name     = customer.get('name', '')
        trial_end = data.get('trial_end')
        
        trial_ends_at = (
            datetime.fromtimestamp(
                trial_end
            ).isoformat()
            if trial_end else ''
        )

        _save_subscriber({
            'email':                  email,
            'name':                   name or '',
            'stripe_customer_id':     data[
                'customer'
            ],
            'stripe_subscription_id': data['id'],
            'status':                 'trial',
            'subscribed_at':          (
                datetime.utcnow().isoformat()
            ),
            'trial_ends_at':          trial_ends_at,
            'cancelled_at':           ''
        })
        logger.info(
            f"[STRIPE] New trial: {email}"
        )

    elif event_type in [
        'invoice.payment_succeeded'
    ]:
        # Trial converted or renewed
        _update_subscriber_status(
            data.get('subscription'), 'active'
        )

    elif event_type in [
        'customer.subscription.deleted',
        'invoice.payment_failed'
    ]:
        # Cancelled or payment failed
        _update_subscriber_status(
            data.get('id') or data.get(
                'subscription'
            ),
            'cancelled'
        )


def _save_subscriber(sub_data):
    """Save new subscriber to CSV."""
    import csv
    FILE = 'mora_assists_subscribers.csv'
    FIELDS = [
        'email', 'name',
        'stripe_customer_id',
        'stripe_subscription_id',
        'status', 'subscribed_at',
        'trial_ends_at', 'cancelled_at'
    ]
    file_exists = os.path.exists(FILE)
    with open(FILE, 'a', newline='') as f:
        writer = csv.DictWriter(
            f, fieldnames=FIELDS
        )
        if not file_exists:
            writer.writeheader()
        writer.writerow(sub_data)


def _update_subscriber_status(
    subscription_id, new_status
):
    """Update subscriber status in CSV."""
    import csv, tempfile, shutil
    FILE = 'mora_assists_subscribers.csv'
    if not os.path.exists(FILE):
        return

    FIELDS = [
        'email', 'name',
        'stripe_customer_id',
        'stripe_subscription_id',
        'status', 'subscribed_at',
        'trial_ends_at', 'cancelled_at'
    ]

    rows = []
    with open(FILE, 'r') as f:
        reader = csv.DictReader(f)
        for row in reader:
            if row.get(
                'stripe_subscription_id'
            ) == subscription_id:
                row['status'] = new_status
                if new_status == 'cancelled':
                    row['cancelled_at'] = (
                        datetime.utcnow()
                        .isoformat()
                    )
            rows.append(row)

    with open(FILE, 'w', newline='') as f:
        writer = csv.DictWriter(
            f, fieldnames=FIELDS
        )
        writer.writeheader()
        writer.writerows(rows)

    logger.info(
        f"[STRIPE] Updated {subscription_id} "
        f"→ {new_status}"
    )

(function() {

  const POPUP_KEY = 'assists_popup_seen';
  
  // Don't show if already seen this session
  if (sessionStorage.getItem(POPUP_KEY)) {
    return;
  }

  // Don't show if email gate is still visible
  const gate = document.getElementById(
    'email-gate'
  );
  const gateVisible = gate && 
    gate.style.display !== 'none' &&
    !localStorage.getItem('mora_access');

  if (gateVisible) return;

  setTimeout(function() {

    // Check gate again at trigger time
    if (!localStorage.getItem('mora_access')) {
      return;
    }

    const popup = document.getElementById(
      'assists-popup'
    );
    if (!popup) return;

    popup.classList.remove('hidden');
    popup.classList.add('flex');
    sessionStorage.setItem(POPUP_KEY, '1');

    // Handle CTA click → Stripe checkout
    document.getElementById('assists-cta')
      ?.addEventListener('click', async () => {
        const btn = document.getElementById(
          'assists-cta'
        );
        btn.textContent = 'Loading...';
        btn.disabled = true;

        const email = localStorage.getItem(
          'mora_email'
        ) || '';

        try {
          const res = await fetch(
            '/assists/checkout',
            {
              method: 'POST',
              headers: {
                'Content-Type': 'application/json'
              },
              body: JSON.stringify({ email })
            }
          );
          const data = await res.json();

          if (data.checkout_url) {
            window.location.href = (
              data.checkout_url
            );
          } else {
            btn.textContent = (
              'Start My 3-Day Free Trial →'
            );
            btn.disabled = false;

            'Start My 3-Day Free Trial →'

          btn.disabled = false;

    // Handle dismiss

        popup.classList.remove('flex');

    });
