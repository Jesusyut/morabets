=== BUG 1: MLB PROPS RETURN VALUE ===

The logs confirm:
"Grouped 181 raw props → 134 unique props" ✅
"Props saved to mlb_props_cache.json ✅" ✅
"[MLB PROPS] Cached 0 props" ← BUG

The function _fetch_and_process_mlb_props()
saves props to file correctly but then
the caller receives 0 props back.

Root cause: the function either does not
return the props list, or the route reads
from cache before the file write completes,
or a variable shadowing issue zeroes it out.

Find _fetch_and_process_mlb_props() in app.py
and replace it with this exact version:

def _fetch_and_process_mlb_props():
    """
    Fetch, group, and cache MLB props.
    Returns the processed props list.
    """
    try:
        from odds_api import (
            fetch_player_props,
            group_props_by_player
        )
        from enrichment import cache_props_to_file

        logger.info("[MLB PROPS] Starting fetch...")

        raw = fetch_player_props()

        if not raw:
            logger.warning(
                "[MLB PROPS] No raw props returned"
            )
            return []

        logger.info(
            f"[MLB PROPS] {len(raw)} raw props "
            f"— grouping now"
        )

        # Group and apply no-vig filter
        props = group_props_by_player(raw)

        logger.info(
            f"[MLB PROPS] {len(props)} props "
            f"after grouping"
        )

        if not props:
            logger.warning(
                "[MLB PROPS] 0 props after grouping"
            )
            return []

        # Save to file cache
        cache_props_to_file(
            props, "mlb_props_cache.json"
        )
        logger.info(
            f"[MLB PROPS] Cached {len(props)} props ✅"
        )

        # CRITICAL: return the list
        return props

    except Exception as e:
        logger.error(
            f"[MLB PROPS] _fetch_and_process "
            f"failed: {e}",
            exc_info=True
        )
        return []

@app.route("/api/mlb/props")
def mlb_props():

        from enrichment import load_props_from_file

        # Check file cache age
        import os
        cache_file = "mlb_props_cache.json"
        cache_fresh = False

        if os.path.exists(cache_file):
            age_seconds = (
                time.time() -
                os.path.getmtime(cache_file)
            )
            # Cache valid for 23 hours
            cache_fresh = age_seconds < 82800

        if cache_fresh:
            props = load_props_from_file(cache_file)
            if props:
                logger.info(
                    f"[MLB PROPS] Serving "
                    f"{len(props)} from cache"
                )
                return jsonify({
                    "props":  props,
                    "count":  len(props),

=== BUG 2: NHL RATE LIMITING ===

The logs show:
"Rate limited — slow down"
"Total raw props: 0"

The sleep between NHL event requests is not
long enough. When a 429 hits, 3 seconds is 
not sufficient — the API needs more time.

Also the current code may be continuing the
loop after a rate limit hit instead of backing
off properly.

In nhl_odds_api.py find fetch_props_for_event()
or the main fetch_nhl_props() loop.

Fix the rate limit handling:

1. Increase sleep between events from 0.5s to 1.5s:

# Between each event (not just between batches)
# Add at the top of the event loop:
time.sleep(1.5)  # Respect rate limits

2. When 429 hits, sleep longer and retry ONCE:

if response.status_code == 429:
    logger.warning(
        "[NHL] Rate limited — sleeping 10s"
    )
    time.sleep(10)  # Back off 10 seconds
    
    # Retry once after backoff
    try:
        response = requests.get(
            url, params=params, timeout=15
        )
        if response.status_code == 429:
            # Still rate limited — skip this event
            logger.warning(
                "[NHL] Still rate limited "
                "— skipping event"
            )
            return []
    except Exception:
        return []

3. Reduce concurrent NHL event fetching.
   Change ThreadPoolExecutor max_workers from
   3 or 5 down to 1 for NHL:

# Sequential fetch for NHL to avoid 429
# max_workers=1 means one event at a time
with ThreadPoolExecutor(max_workers=1) as executor:
    results = list(
        executor.map(fetch_props_for_event, events)
    )

This is slower but NHL has fewer games per day
(4-8 typical) so sequential is fine and
eliminates the rate limit problem entirely.

4. Add a delay between events in the sequential
   fetch by wrapping fetch_props_for_event:

def fetch_props_for_event_with_delay(event):

=== BUG 3: NHL PROPS "NOT POSTED YET" ===

The logs show ALL NHL events returning 422
"Props not posted yet" — this is normal when:
- Games are more than ~24 hours away
- Books haven't released props yet for that day

The fix: when all events return 422, serve 
the previous day's cached props if they exist
rather than returning empty.

In _fetch_and_process_nhl_props() add fallback:

def _fetch_and_process_nhl_props():
    try:
        from nhl_odds_api import fetch_nhl_props
        from odds_api import group_props_by_player
        from enrichment import (
            cache_props_to_file,
            load_props_from_file
        )

        logger.info("[NHL PROPS] Starting fetch...")

        raw = fetch_nhl_props()

        if not raw:
            logger.warning(
                "[NHL PROPS] No raw props — "
                "checking yesterday's cache"
            )
            # Serve stale cache rather than empty
            stale = load_props_from_file(
                "nhl_props_cache.json"
            )
            if stale:
                logger.info(
                    f"[NHL PROPS] Serving "
                    f"{len(stale)} stale props "
                    f"(no fresh data available)"
                )
            return stale or []

        props = group_props_by_player(raw)

        if not props:
            logger.warning(
                "[NHL PROPS] 0 after grouping"
            )
            return []

        cache_props_to_file(
            props, "nhl_props_cache.json"
        )
        logger.info(
            f"[NHL PROPS] Cached {len(props)} ✅"
        )

        # CRITICAL: return the list
        return props

    except Exception as e:
        logger.error(
            f"[NHL PROPS] failed: {e}",
            exc_info=True
        )
        return []

async function loadSportBoard(sport) {
  try {
    showLoadingSkeleton();
    const res = await fetch(
      `/api/${sport}/props?t=${Date.now()}`
    );
    const data = await res.json();

    // MUST read data.props not data
    const props = data.props || data.picks || [];
    
    console.log(
      `[BOARD] ${sport} props received:`,
      props.length
    );

    if (props.length === 0) {
      showEmptyState(sport);
      return;
    }

    renderBoard(props, sport);

  } catch(err) {
    console.error('[BOARD] Load error:', err);
    showErrorState(sport);
  }
}

@app.route("/api/cache-check")
def cache_check():
    """
    Quick check of what's in prop cache files.
    Tells you count, age, and sample prop.
    """
    import os
    from enrichment import load_props_from_file

    def file_info(filename):
        if not os.path.exists(filename):
            return {
                "exists": False,
                "count":  0,
                "age_minutes": None,
                "sample": None
            }
        age = (
            time.time() -
            os.path.getmtime(filename)
        ) / 60
        props = load_props_from_file(filename)
        return {
            "exists":      True,
            "count":       len(props),
            "age_minutes": round(age, 1),
            "sample":      props[0] if props else None,
            "tiers": {
                "LOCK": len([
                    p for p in props
                    if p.get("confidence_tier")
                    == "LOCK"
                ]),
                "FIRE": len([
                    p for p in props
                    if p.get("confidence_tier")
                    == "FIRE"
                ]),
                "LOW": len([
                    p for p in props
                    if p.get("confidence_tier")
                    == "LOW"
                ])
            }
        }

    return jsonify({
        "mlb": file_info("mlb_props_cache.json"),
        "nhl": file_info("nhl_props_cache.json"),
        "timestamp": datetime.utcnow().isoformat()
    })
