=== THE CORE PROBLEM TO FIX ===

Current broken flow:
  3 books → remove vig → fair probability → surface as pick

Why it fails:
  - No-vig from the same 3 books just recreates 
    market opinion
  - 60% fair probability means nothing if the 
    offered line is -175 (break-even = 63.6%)
  - That is a NEGATIVE EV bet the current system 
    calls a confident pick
  - There is no comparison between fair price 
    and offered price anywhere in the code

Correct flow to build:
  many books → weighted fair probability → 
  fair odds → compare to best offered line → 
  EV% calculation → threshold filter by market type → 
  surface only positive EV picks → log everything 
  for CLV tracking

def american_to_implied(american_odds: int) -> float:
    """Convert American odds to raw implied probability."""
    if american_odds > 0:
        return 100 / (american_odds + 100)
    else:
        return abs(american_odds) / (abs(american_odds) + 100)


def implied_to_american(prob: float) -> int:
    """Convert probability back to American odds."""
    if prob <= 0 or prob >= 1:
        return 0
    if prob >= 0.5:
        return round(-(prob / (1 - prob)) * 100)
    else:
        return round(((1 - prob) / prob) * 100)


def remove_vig_two_sided(
    over_price: int, 
    under_price: int
) -> tuple[float, float]:
    """
    Remove vig from a two-sided market.
    Returns (no_vig_over_prob, no_vig_under_prob).
    """
    raw_over = american_to_implied(over_price)
    raw_under = american_to_implied(under_price)
    total = raw_over + raw_under
    if total == 0:
        return 0.5, 0.5
    return raw_over / total, raw_under / total


def calculate_weighted_fair_probability(
    book_prices: list[dict],
    side: str = "over"
) -> tuple[Optional[float], int, float]:
    """
    Calculate weighted consensus fair probability
    using all available books.
    
    book_prices: list of {
        'book': str,
        'over_price': int,
        'under_price': int
    }
    side: 'over' or 'under'
    
    Returns:
        (fair_probability, book_count, total_weight)
        fair_probability is None if insufficient data
    """
    weighted_prob_sum = 0.0
    total_weight = 0.0
    book_count = 0
    
    for entry in book_prices:
        book_name = entry.get('book', '').lower()
        over_p = entry.get('over_price')
        under_p = entry.get('under_price')
        
        if over_p is None or under_p is None:
            continue
        
        no_vig_over, no_vig_under = remove_vig_two_sided(
            over_p, under_p
        )
        
        prob = no_vig_over if side == "over" else no_vig_under
        weight = BOOK_WEIGHTS.get(book_name, 1.0)
        
        weighted_prob_sum += prob * weight
        total_weight += weight
        book_count += 1
    
    if book_count == 0 or total_weight == 0:
        return None, 0, 0.0
    
    fair_prob = weighted_prob_sum / total_weight
    return round(fair_prob, 4), book_count, total_weight


def calculate_ev(
    fair_probability: float,
    offered_odds: int
) -> float:
    """
    Calculate Expected Value percentage.
    
    EV% = (fair_prob × profit_if_win) - 
          (fair_prob_lose × stake)
    
    Normalized to percentage of stake.
    
    Positive EV = bet has mathematical edge.
    Negative EV = book has edge over you.
    
    Example:
      fair_prob = 0.60 (60%)
      offered_odds = -175
      break_even = 0.6364 (63.64%)
      EV = (0.60 × 0.5714) - (0.40 × 1.0) = -0.057 = -5.7%
      This is a NEGATIVE EV bet despite 60% fair prob.
    """
    if offered_odds > 0:
        # Underdog: profit = odds/100 per unit staked
        profit_per_unit = offered_odds / 100
    else:
        # Favorite: profit = 100/|odds| per unit staked
        profit_per_unit = 100 / abs(offered_odds)
    
    lose_prob = 1 - fair_probability
    
    ev = (fair_probability * profit_per_unit) - (
        lose_prob * 1.0
    )
    return round(ev * 100, 2)  # as percentage


def calculate_edge_in_cents(
    fair_probability: float,
    offered_odds: int
) -> float:
    """
    Edge in cents on the dollar.
    
    Converts fair probability to American odds,
    then measures difference from offered odds
    in implied probability terms.
    
    Positive = offered odds are BETTER than fair.
    Negative = offered odds are WORSE than fair.
    """
    fair_american = implied_to_american(fair_probability)
    fair_implied = american_to_implied(fair_american)
    offered_implied = american_to_implied(offered_odds)
    
    # Edge = how much better your price is vs fair
    edge = fair_implied - offered_implied
    return round(edge * 100, 2)  # in percentage points


def get_best_offered_line(
    book_prices: list[dict],
    side: str = "over"
) -> tuple[Optional[int], Optional[str]]:
    """
    Find the best (most favorable) offered odds
    across all available books for the target side.
    
    Best odds = highest American odds value
    (e.g. -105 is better than -125, +110 better than -110)
    """
    best_price = None
    best_book = None
    
    for entry in book_prices:
        price_key = (
            'over_price' if side == 'over' 
            else 'under_price'
        )
        price = entry.get(price_key)
        book = entry.get('book', '')
        
        if price is None:
            continue
        
        if best_price is None or price > best_price:
            best_price = price
            best_book = book
    
    return best_price, best_book


def evaluate_pick(
    player_or_team: str,
    market_type: str,
    side: str,
    line: Optional[float],
    book_prices: list[dict],
    game_time: Optional[str] = None,
    minutes_to_game: Optional[int] = None
) -> dict:
    """
    Full EV evaluation of a potential bet.
    
    Returns a complete pick evaluation dict with:
    - fair_probability (weighted consensus)
    - fair_odds (American)
    - best_offered_odds and book
    - ev_pct (Expected Value %)
    - edge_pct (edge in probability points)
    - passes_threshold (bool)
    - rejection_reason (if rejected)
    - confidence tier based on EV, not just probability
    """
    result = {
        "player":           player_or_team,
        "market_type":      market_type,
        "side":             side,
        "line":             line,
        "book_count":       0,
        "fair_probability": None,
        "fair_odds":        None,
        "best_offered_odds": None,
        "best_book":        None,
        "break_even_prob":  None,
        "ev_pct":           None,
        "edge_pct":         None,
        "passes_threshold": False,
        "rejection_reason": None,
        "confidence_tier":  "LOW",
        "game_time":        game_time,
        "minutes_to_game":  minutes_to_game,
        "evaluated_at":     int(time.time()),
        "all_books":        book_prices
    }
    
    # ── Step 1: Check minimum book requirement ──────────
    min_books = MIN_BOOKS_REQUIRED.get(
        market_type, 
        MIN_BOOKS_REQUIRED["default"]
    )
    
    valid_books = [
        b for b in book_prices
        if b.get('over_price') is not None 
        and b.get('under_price') is not None
    ]
    
    result["book_count"] = len(valid_books)
    
    if len(valid_books) < min_books:
        result["rejection_reason"] = (
            f"Insufficient books: {len(valid_books)} "
            f"available, {min_books} required"
        )
        return result
    
    # ── Step 2: Calculate weighted fair probability ─────
    fair_prob, book_count, weight = (
        calculate_weighted_fair_probability(
            valid_books, side
        )
    )
    
    if fair_prob is None:
        result["rejection_reason"] = (
            "Could not calculate fair probability"
        )
        return result
    
    result["fair_probability"] = fair_prob
    result["fair_odds"] = implied_to_american(fair_prob)
    
    # ── Step 3: Find best offered line ──────────────────
    best_odds, best_book = get_best_offered_line(
        valid_books, side
    )
    
    if best_odds is None:
        result["rejection_reason"] = (
            "No offered line found"
        )
        return result
    
    result["best_offered_odds"] = best_odds
    result["best_book"] = best_book
    result["break_even_prob"] = round(
        american_to_implied(best_odds), 4
    )
    
    # ── Step 4: Calculate EV and edge ───────────────────
    ev = calculate_ev(fair_prob, best_odds)
    edge = calculate_edge_in_cents(fair_prob, best_odds)
    
    result["ev_pct"] = ev
    result["edge_pct"] = edge
    
    # ── Step 5: Apply EV threshold filter ───────────────
    threshold = EV_THRESHOLDS.get(
        market_type,
        EV_THRESHOLDS["default"]
    )
    
    if ev < threshold:
        result["rejection_reason"] = (
            f"EV {ev:.1f}% below threshold "
            f"{threshold:.1f}% for {market_type}"
        )
        # Still return the data — just not a bet signal
        return result
    
    # ── Step 6: Assign confidence tier by EV ───────────
    # EV-based tiers are more honest than prob-based
    result["passes_threshold"] = True
    
    if ev >= 6.0:
        result["confidence_tier"] = "LOCK"
    elif ev >= 3.0:
        result["confidence_tier"] = "FIRE"
    else:
        result["confidence_tier"] = "EDGE"
    
    logger.info(
        f"[EV] ✅ {player_or_team} {market_type} "
        f"{side} | Fair: {result['fair_odds']} | "
        f"Best: {best_odds}@{best_book} | "
        f"EV: {ev:.1f}% | Edge: {edge:.1f}%"
    )
    
    return result

import json

def log_bet_entry(
    event_id: str,
    player_or_team: str,
    market_type: str,
    side: str,
    line,
    book: str,

        "logged_at":        int(time.time()),
        "logged_at_iso":    datetime.utcnow().isoformat(),
        "player_or_team":   player_or_team,
        "market_type":      market_type,
        "side":             side,
        "line":             line,

    }

        f"{market_type} @ {offered_odds}"
    )
    return entry


def update_closing_line(
    event_id: str,
    market_type: str,
    side: str,
    closing_odds: int
):
    """
    Update a logged bet with the closing line.
    CLV = entry odds vs closing odds (in prob points).
    Positive CLV = you got a better price than close.
    """
    log = _load_log()
    
    for entry in log:
        if (entry["id"] == str(event_id) and
                entry["market_type"] == market_type and
                entry["side"] == side and
                entry["status"] == "open"):
            
            from ev_engine import (
                american_to_implied,
                calculate_edge_in_cents
            )
            
            entry_implied = american_to_implied(
                entry["offered_odds"]
            )
            close_implied = american_to_implied(
                closing_odds
            )
            
            # Positive = your entry was better than close
            clv = round(
                (close_implied - entry_implied) * 100, 2
            )
            
            entry["closing_odds"] = closing_odds
            entry["clv"] = clv
            
            logger.info(
                f"[CLV] Updated close for {entry['player_or_team']}: "
                f"entry={entry['offered_odds']}, "
                f"close={closing_odds}, CLV={clv:+.2f}%"
            )
    
    _save_log(log)


def record_result(
    event_id: str,
    market_type: str,
    side: str,
    result: str  # "W", "L", or "P" (push)
):
    """Record the actual outcome of a bet."""
    log = _load_log()
    
    for entry in log:
        if (entry["id"] == str(event_id) and
                entry["market_type"] == market_type and
                entry["side"] == side):
            entry["result"] = result
            entry["status"] = "settled"
    
    _save_log(log)


def get_performance_report() -> dict:
    """
    Generate performance metrics from the CLV log.
    This is how you know if the system is actually working.
    """
    log = _load_log()
    settled = [e for e in log if e["status"] == "settled"]
    
    if not settled:
        return {"message": "No settled bets yet"}
    
    total = len(settled)
    wins = len([e for e in settled if e["result"] == "W"])
    losses = len([e for e in settled if e["result"] == "L"])
    
    # CLV stats (only for entries with closing line)
    clv_entries = [
        e for e in settled if e["clv"] is not None
    ]
    avg_clv = (
        sum(e["clv"] for e in clv_entries) / len(clv_entries)
        if clv_entries else None
    )
    
    # ROI calculation (flat stakes assumed)
    roi_points = 0
    for e in settled:
        if e["result"] == "W":
            odds = e["offered_odds"]
            if odds > 0:
                roi_points += odds / 100
            else:
                roi_points += 100 / abs(odds)
        elif e["result"] == "L":
            roi_points -= 1.0
    
    roi_pct = round((roi_points / total) * 100, 2)
    
    # Break down by market type
    by_market = {}
    market_types = set(e["market_type"] for e in settled)
    for mt in market_types:
        mt_entries = [
            e for e in settled 
            if e["market_type"] == mt
        ]
        mt_wins = len([
            e for e in mt_entries 
            if e["result"] == "W"
        ])
        by_market[mt] = {
            "count":    len(mt_entries),
            "wins":     mt_wins,
            "win_rate": round(
                mt_wins / len(mt_entries) * 100, 1
            )
        }
    
    return {
        "total_bets":       total,
        "wins":             wins,
        "losses":           losses,
        "win_rate":         round(wins/total*100, 1),
        "roi_pct":          roi_pct,
        "avg_clv":          avg_clv,
        "avg_ev_at_entry":  round(
            sum(e["ev_pct"] for e in settled 
                if e["ev_pct"]) / total, 2
        ),
        "by_market":        by_market,
        "clv_positive":     len([
            e for e in clv_entries if e["clv"] > 0
        ]),
        "clv_negative":     len([
            e for e in clv_entries if e["clv"] < 0
        ])
    }


def _load_log() -> list:
    try:
        with open(CLV_LOG_FILE, 'r') as f:
            return json.load(f)
    except FileNotFoundError:
        return []
    except Exception:
        return []


def _save_log(log: list):
    try:
        with open(CLV_LOG_FILE, 'w') as f:
            json.dump(log, f, indent=2)
    except Exception as e:
        logger.error(f"[CLV] Save failed: {e}")

@app.route("/api/performance")
def performance_report():
    """
    Internal performance tracking endpoint.
    Shows ROI, CLV, win rate by market type.
    No auth required for now — add later.
    """
    from clv_tracker import get_performance_report
    return jsonify(get_performance_report())

@app.route("/api/performance/log")
def clv_log():
    """Return raw CLV log for inspection."""
    from clv_tracker import _load_log
    log = _load_log()
    # Show most recent 50 entries
    return jsonify({
        "count": len(log),
        "recent": log[-50:],
        "open": len([
            e for e in log 
            if e.get("status") == "open"
        ]),
        "settled": len([
            e for e in log 
            if e.get("status") == "settled"
        ])
    })
