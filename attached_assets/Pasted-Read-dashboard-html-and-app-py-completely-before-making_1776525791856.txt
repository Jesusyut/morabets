=== JAVASCRIPT LOGIC SUMMARY ===

On page load run this sequence:

const visited = localStorage.getItem(
  'mora_visited'
)
const fridayCaptured = localStorage.getItem(
  'mora_friday_captured'  
)
const fridayDismissed = localStorage.getItem(
  'mora_friday_dismissed'
)
const assistsLast = localStorage.getItem(
  'mora_assists_popup_last'
)
const today = new Date().getDay()
// 5 = Friday

// App install toast — first visit only
if (!visited) {
  setTimeout(showAppInstallToast, 10000)
  localStorage.setItem('mora_visited','true')
}

// Friday modal logic
const isFriday = today === 5
const fridayAlreadyHandled = 
  fridayCaptured || fridayDismissed

if (isFriday && !fridayAlreadyHandled) {
  setTimeout(showFridayModal, 45000)
}

// Mora Assists popup logic
const assistsDelay = fridayCaptured 
  ? 60000 : 90000
const oneDayAgo = Date.now() - 86400000
const assistsRecent = assistsLast && 
  parseInt(assistsLast) > oneDayAgo

if (!assistsRecent) {
  if (!isFriday || fridayAlreadyHandled) {
    setTimeout(
      showMoraAssistsPopup,
      assistsDelay
    )
  } else {
    // Friday modal shown — 
    // wait for next session
    // do nothing this session
  }
}
