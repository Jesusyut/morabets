=== STEP 2: UPDATE /api/gate-signup 
IN app.py ===

Find the gate-signup route.

Add the Zapier webhook fire FIRST

import requests as http_requests

@app.route('/api/gate-signup', methods=['POST'])
def gate_signup():
    try:
        data  = request.json or {}
        email = data.get(
            'email', ''
        ).strip().lower()
        name  = data.get('name', '').strip()

        if not email or '@' not in email:
            return jsonify({
                "success": False,
                "error":   "Invalid email"
            }), 400

        # ── ZAPIER WEBHOOK ────────────────
        # Fires first — before any local 
        # storage attempt. Email is safe
        # in external database immediately.
        zapier_url = os.environ.get(
            'ZAPIER_WEBHOOK_URL'
        )
        if zapier_url:
            try:
                hook_resp = http_requests.post(
                    zapier_url,
                    json={
                        "email":     email,
                        "name":      name,
                        "source":    "mora_bets_gate",
                        "signed_up": datetime.utcnow(
                        ).isoformat(),
                        "url":       "morabets.com"
                    },
                    timeout=5
                )
                logger.info(
                    f"[ZAPIER] ✅ Fired for "
                    f"{email} — "
                    f"status {hook_resp.status_code}"
                )
            except Exception as ze:
                # Never block signup for 
                # webhook failure
                # Email still saves to CSV
                logger.error(
                    f"[ZAPIER] ❌ Failed "
                    f"for {email}: {ze}"
                )
        else:
            logger.warning(
                "[ZAPIER] No webhook URL set"
            )
        # ── END ZAPIER ────────────────────

        # Check if already exists in CSV
        existing = _load_emails()
        if email in existing:
            logger.info(
                f"[GATE] Returning: {email}"
            )
            return jsonify({
                "success":  True,
                "existing": True
            })

        # Save to local CSV backup
        _save_email(email, name)

        count = len(_load_emails())
        logger.info(
            f"[GATE] ✅ Saved: {email} "
            f"(total: {count})"
        )

        # Fire Meta Pixel server-side Lead
        try:
            from meta_pixel import (
                track_lead,
                get_event_id
            )
            event_id = get_event_id()
            track_lead(
                request,
                customer_email=email,
                event_id=event_id
            )
        except Exception as pe:
            logger.warning(
                f"[META] Lead failed: {pe}"
            )

        return jsonify({
            "success":  True,
            "count":    count
        })

    except Exception as e:
        logger.error(
            f"[GATE] Route error: {e}"
        )
        # Always return success —
        # never block user from dashboard
        return jsonify({"success": True})
