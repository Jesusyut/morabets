--- TIMING LOGIC ---

// Show once per 24 hours
const last = localStorage.getItem(
  'mora_assists_last_shown'
)
const oneDayAgo = Date.now() - 86400000
const shouldShow = !last || 
  parseInt(last) < oneDayAgo

if (shouldShow) {
  setTimeout(() => {
    document.getElementById(
      'mora-assists-modal'
    ).classList.remove('hidden')
  }, 45000)
}

function closeMoraModal() {
  document.getElementById(
    'mora-assists-modal'
  ).classList.add('hidden')
  localStorage.setItem(
    'mora_assists_last_shown',
    Date.now().toString()
  )
}
