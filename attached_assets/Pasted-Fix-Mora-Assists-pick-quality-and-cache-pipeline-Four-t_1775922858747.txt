def load_full_board():
    """
    Load all cached props and lines
    across every sport.
    Dynamically finds cache files
    wherever they actually live.
    """
    from enrichment import load_props_from_file

    board = {
        "props":        [],
        "lines":        [],
        "sports_found": []
    }

    # ── PROPS ──────────────────────────────
    # Check both /var/data/ and local paths
    prop_files = [
        '/var/data/mlb_props_cache.json',
        '/var/data/nhl_props_cache.json',
        '/var/data/nfl_props_cache.json',
        'mlb_props_cache.json',
        'nhl_props_cache.json',
        'nfl_props_cache.json',
    ]

    for f in prop_files:
        try:
            props = load_props_from_file(f)
            if props:
                board["props"].extend(props)
                sport = (
                    'MLB' if 'mlb' in f else
                    'NHL' if 'nhl' in f else
                    'NFL' if 'nfl' in f else ''
                )
                if sport and sport not in (
                    board["sports_found"]
                ):
                    board["sports_found"].append(
                        sport
                    )
                logger.info(
                    f"[BOARD] {len(props)} props "
                    f"from {f}"
                )
        except Exception:
            pass

    # ── LINES — read from Redis/memory cache ──
    # Lines live in the app memory cache
    # not in separate files
    try:
        from app import cache_get
        import json as _json

        line_keys = {
            'MLB': [
                'mlb_odds',
                'mlb_curated_picks',
                'mlb_enriched_odds'
            ],
            'NHL': [
                'nhl_odds',
                'nhl_curated_picks',
                'nhl_enriched_odds'
            ]
        }

        for sport, keys in line_keys.items():
            for key in keys:
                try:
                    val = cache_get(key)
                    if val:
                        data = (
                            _json.loads(val)
                            if isinstance(val, str)
                            else val
                        )
                        if isinstance(data, list) and data:
                            board["lines"].extend(data)
                            if sport not in (
                                board["sports_found"]
                            ):
                                board[
                                    "sports_found"
                                ].append(sport)
                            logger.info(
                                f"[BOARD] {len(data)}"
                                f" lines from "
                                f"cache key: {key}"
                            )
                            break
                except Exception:
                    continue

    except Exception as e:
        logger.warning(
            f"[BOARD] Lines cache error: {e}"
        )

    # ── FALLBACK: use props as anchor source ──
    # If no lines found, LLM will use high-prob
    # props as anchors per the updated prompt
    if not board["lines"]:
        logger.warning(
            "[BOARD] No lines cache found. "
            "LLM will select anchors from "
            "high-probability props."
        )

    # Filter props to quality minimum
    board["props"] = [
        p for p in board["props"]
        if p.get("no_vig_prob", 0) >= 52
    ]

    logger.info(
        f"[BOARD] Final: "
        f"{len(board['props'])} props, "
        f"{len(board['lines'])} lines, "
        f"sports: {board['sports_found']}"
    )
    return board

def validate_picks(picks_data):
    """
    Remove any picks that violate 
    hard rules before sending email.
    Prevents -500 picks from going out.
    """
    if not picks_data:
        return picks_data

    clean_picks = []
    removed = 0

    for p in picks_data.get('picks', []):
        odds = p.get('odds', 0)
        prob = p.get('no_vig_prob', 0)
        team = p.get('team', '')
        player = p.get('player', '')

        # Hard reject: no data
        if odds == 0 and prob == 0.0:
            logger.warning(
                f"[VALIDATE] Removed empty "
                f"pick {p.get('pick_number')}"
            )
            removed += 1
            continue

        # Hard reject: juice worse than -250
        if odds < -250:
            logger.warning(
                f"[VALIDATE] Removed pick "
                f"{p.get('pick_number')} — "
                f"juice {odds} exceeds -250 limit"
            )
            removed += 1
            continue

        # Hard reject: probability above 70%
        if prob > 70.0:
            logger.warning(
                f"[VALIDATE] Removed pick "
                f"{p.get('pick_number')} — "
                f"prob {prob}% above 70% ceiling"
            )
            removed += 1
            continue

        # Hard reject: probability below 60%
        if prob < 60.0 and p['type'] != 'casual':
            logger.warning(
                f"[VALIDATE] Removed pick "
                f"{p.get('pick_number')} — "
                f"prob {prob}% below 60% floor"
            )
            removed += 1
            continue

        clean_picks.append(p)

    if removed > 0:
        logger.warning(
            f"[VALIDATE] Removed {removed} "
            f"picks that failed quality check"
        )

    picks_data['picks'] = clean_picks
    return picks_data


Call validate_picks() in run_daily_assists()
