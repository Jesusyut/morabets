<!-- OVERLAY JAVASCRIPT -->
<script>
(function() {
  var seen = localStorage.getItem(
    'mora_intro_seen'
  );

  if (!seen) {
    var overlay = document.getElementById(
      'mora-intro-overlay'
    );
    if (overlay) {
      overlay.style.display = 'flex';
    }
  }
})();

function enterDashboard() {
  localStorage.setItem(
    'mora_intro_seen', 'true'
  );
  var overlay = document.getElementById(
    'mora-intro-overlay'
  );
  if (overlay) {
    overlay.style.opacity = '0';
    overlay.style.transition =
      'opacity 0.3s ease';
    setTimeout(function() {
      overlay.style.display = 'none';
    }, 300);
  }
}
</script>
