AS ACTIVE SUBSCRIBER ===

Run this in the Replit shell to add
the subscriber directly to the CSV:

python3 -c "
import csv, os
from datetime import datetime

FILE = 'mora_assists_subscribers.csv'
FIELDS = [
    'email', 'name',
    'stripe_customer_id',
    'stripe_subscription_id',
    'status', 'subscribed_at',
    'trial_ends_at', 'cancelled_at'
]

file_exists = os.path.exists(FILE)

with open(FILE, 'a', newline='') as f:
    writer = csv.DictWriter(
        f, fieldnames=FIELDS
    )
    if not file_exists:
        writer.writeheader()
    writer.writerow({
        'email':                  'jesusm433@icloud.com',
        'name':                   'Jesus',
        'stripe_customer_id':     'manual_test',
        'stripe_subscription_id': 'manual_test',
        'status':                 'active',
        'subscribed_at':          datetime.utcnow().isoformat(),
        'trial_ends_at':          '',
        'cancelled_at':           ''
    })

print('✅ jesusm433@icloud.com added as active subscriber')

# Verify
with open(FILE) as f:
    rows = list(csv.DictReader(f))
print(f'Total subscribers: {len(rows)}')
for r in rows:
    print(f'  {r[\"email\"]} — {r[\"status\"]}')
"
