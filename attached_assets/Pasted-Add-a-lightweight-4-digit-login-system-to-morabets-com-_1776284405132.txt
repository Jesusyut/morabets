@app.route('/api/send-code', methods=['POST'])
def send_login_code():
    import random
    from datetime import datetime, timedelta

    email = (
        request.json.get('email', '')
        .lower().strip()
    )
    if not email or '@' not in email:
        return jsonify(
            {'error': 'Invalid email'}
        ), 400

    code = str(random.randint(1000, 9999))
    expiry = (
        datetime.utcnow() + timedelta(minutes=10)
    ).isoformat()

    # Store code in memory cache
    stored_value = f'{code}|{expiry}'
    try:
        cache_set(
            f'code_{email}',
            stored_value,
            ex=600
        )
    except Exception:
        if not hasattr(app, '_login_codes'):
            app._login_codes = {}
        app._login_codes[email] = stored_value

    # Send via SendGrid
    try:
        import sendgrid
        from sendgrid.helpers.mail import Mail
        sg = sendgrid.SendGridAPIClient(
            api_key=os.environ.get('SENDGRID_API_KEY')
        )
        msg = Mail(
            from_email=os.environ.get(
                'EMAIL_FROM',
                'picks@morabets.com'
            ),
            to_emails=email,
            subject=f'Your Mora Bets code: {code}',
            plain_text_content=(
                f'Your code: {code}\n\n'
                f'Valid for 10 minutes.\n\n'
                f'— Mora Bets\n'
                f'morabets.com'
            )
        )
        sg.send(msg)
        logger.info(f'[LOGIN] Code sent: {email}')
        return jsonify({'success': True})
    except Exception as e:
        logger.error(f'[LOGIN] Send failed: {e}')
        return jsonify(
            {'error': 'Could not send code'}
        ), 500


@app.route('/api/verify-code', methods=['POST'])
def verify_login_code():
    from datetime import datetime
    import csv

    email = (
        request.json.get('email', '')
        .lower().strip()
    )
    code = request.json.get('code', '').strip()

    if not email or not code:
        return jsonify(
            {'error': 'Missing fields'}
        ), 400

    # Retrieve stored code
    stored = None
    try:
        stored = cache_get(f'code_{email}')
    except Exception:
        pass

    if not stored and hasattr(app, '_login_codes'):
        stored = app._login_codes.get(email)

    if not stored:
        return jsonify(
            {'error': 'Code expired — '
                      'request a new one'}
        ), 400

    stored_code, expiry_str = stored.split('|')
    expiry = datetime.fromisoformat(expiry_str)

    if datetime.utcnow() > expiry:
        return jsonify(
            {'error': 'Code expired — '
                      'request a new one'}
        ), 400

    if code != stored_code:
        return jsonify(
            {'error': 'Invalid code'}
        ), 400

    # Valid — determine plan from CSV sources
    plan         = 'free'
    name         = ''
    mora_status  = 'none'
    trial_ends   = ''

    # Check Mora Assists subscribers first
    SUB_FILE = (
        '/var/data/mora_assists_subscribers.csv'
    )
    if os.path.exists(SUB_FILE):
        with open(SUB_FILE) as f:
            for row in csv.DictReader(f):
                if row.get(
                    'email', ''
                ).lower() == email:
                    mora_status = row.get(
                        'status', 'none'
                    )
                    trial_ends = row.get(
                        'trial_ends_at', ''
                    )[:10]
                    name = row.get('name', '')
                    if mora_status in [
                        'active', 'trial'
                    ]:
                        plan = mora_status
                    break

    # Check free email list for name
    if not name:
        FREE_FILE = (
            '/var/data/email_subscribers.csv'
        )
        if os.path.exists(FREE_FILE):
            with open(FREE_FILE) as f:
                for row in csv.DictReader(f):
                    if row.get(
                        'email', ''
                    ).lower() == email:
                        name = row.get('name', '')
                        break

    # Clear used code
    try:
        cache_set(f'code_{email}', '', ex=1)
        if hasattr(app, '_login_codes'):
            app._login_codes.pop(email, None)
    except Exception:
        pass

    logger.info(
        f'[LOGIN] Verified: {email} '
        f'plan={plan}'
    )

    return jsonify({
        'success':     True,
        'email':       email,
        'name':        name,
        'plan':        plan,
        'mora_status': mora_status,
        'trial_ends':  trial_ends
    })

  <script>
  let userEmail = ''

  async function sendCode() {
    const emailInput = document.getElementById(
      'email-input'
    )
    userEmail = emailInput.value.trim()
    if (!userEmail) return
    const btn = document.getElementById('send-btn')
    btn.textContent = 'Sending...'
    btn.disabled = true
    try {
      const res = await fetch('/api/send-code', {
        method: 'POST',
        headers: {'Content-Type':'application/json'},
        body: JSON.stringify({email: userEmail})
      })
      const data = await res.json()
      if (data.success) {
        document.getElementById(
          'step-email'
        ).classList.add('hidden')
        document.getElementById(
          'step-code'
        ).classList.remove('hidden')
        document.getElementById(
          'sent-to'
        ).textContent = `Code sent to ${userEmail}`
      } else {
        btn.textContent = 'Send my code →'
        btn.disabled = false
        alert(data.error || 'Could not send code')
      }
    } catch(e) {
      btn.textContent = 'Send my code →'
      btn.disabled = false
    }
  }

  async function verifyCode() {
    const code = document.getElementById(
      'code-input'
    ).value.toString().trim()
    if (code.length < 4) return
    const btn = document.getElementById('verify-btn')
    btn.textContent = 'Verifying...'
    btn.disabled = true
    try {
      const res = await fetch('/api/verify-code', {
        method: 'POST',
        headers: {'Content-Type':'application/json'},
        body: JSON.stringify({
          email: userEmail, code: code
        })
      })
      const data = await res.json()
      if (data.success) {
        document.getElementById(
          'step-code'
        ).classList.add('hidden')
        document.getElementById(
          'step-dashboard'
        ).classList.remove('hidden')
        localStorage.setItem('mora_email', data.email)
        localStorage.setItem('mora_plan', data.plan)
        localStorage.setItem('mora_name', data.name)
        document.getElementById(
          'user-name'
        ).textContent = data.name || 'there'
        const planId = 'plan-' + (
          data.plan === 'active' ? 'active' :
          data.plan === 'trial'  ? 'trial'  : 'free'
        )
        document.getElementById(
          planId
        )?.classList.remove('hidden')
        const badgeId = 'badge-' + (
          data.plan === 'active'  ? 'active' :
          data.plan === 'trial'   ? 'trial'  :
          data.mora_status === 'expired' 
            ? 'expired' : 'free'
        )
        document.getElementById(
          badgeId
        )?.classList.remove('hidden')
        if (data.trial_ends) {
          ['trial-ends-date',
           'trial-ends-date-2'].forEach(id => {
            const el = document.getElementById(id)
            if (el) el.textContent = data.trial_ends
          })
        }
      } else {
        document.getElementById(
          'code-error'
        ).textContent = data.error
        btn.textContent = 'Verify →'
        btn.disabled = false
      }
    } catch(e) {
      btn.textContent = 'Verify →'
      btn.disabled = false
    }
  }

  function saveSmsOpt() {
    const checked = document.getElementById(
      'sms-opt-in'
    ).checked
    localStorage.setItem(
      'mora_sms_opt', checked ? 'yes' : 'no'
    )
  }

  window.addEventListener('load', () => {

    }

    if (smsOpt === 'yes') {
      document.getElementById(
        'sms-opt-in'
      ).checked = true
    }
