import os
from datetime import datetime

def _cache_is_fresh(filepath, max_hours=20):
    if not os.path.exists(filepath):
        return False
    age = (
        datetime.utcnow().timestamp() - 
        os.path.getmtime(filepath)
    ) / 3600
    return age < max_hours

# MLB props — only fetch if stale
mlb_cache = '/var/data/mlb_props_cache.json'
if _cache_is_fresh(mlb_cache, max_hours=20):
    logger.info(
        f"[CACHE] MLB props fresh — skipping fetch"
    )
else:
    try:
        update_player_props()
        logger.info("✅ MLB props primed")
    except Exception as e:
        logger.warning(f"MLB props failed: {e}")

# NHL props — only fetch if stale
nhl_cache = '/var/data/nhl_props_cache.json'

    )
else:
    try:
        update_nhl_props()
        logger.info("✅ NHL props primed")
    except Exception as e:
        logger.warning(f"NHL props failed: {e}")

# Odds — only fetch if older than 6 hours
odds_flag = '/var/data/odds_last_fetch.txt'

if os.path.exists(odds_flag):
    with open(odds_flag) as f:

AND AUTO-REMOVAL ===

def load_active_subscribers():
    import csv
    from datetime import datetime

    FILE = '/var/data/mora_assists_subscribers.csv'
    FIELDS = [
        'email', 'name',
        'stripe_customer_id',
        'stripe_subscription_id',
        'status', 'subscribed_at',
        'trial_ends_at', 'cancelled_at',
        'phone', 'sms_opt_in'
    ]

    if not os.path.exists(FILE):
        logger.warning('[ASSISTS] No subscriber file')
        return []

    with open(FILE, 'r') as f:
        rows = list(csv.DictReader(f))

    now = datetime.utcnow()
    seen_emails = {}
    updated = False
    active = []

    for row in rows:
        email = row.get('email', '').lower().strip()
        status = row.get('status', '')

        if not email:
            continue

        # Auto-expire trials past end date
        if status == 'trial':
            trial_end_str = row.get(
                'trial_ends_at', ''
            )
            if trial_end_str:
                try:
                    trial_end = datetime.fromisoformat(
                        trial_end_str[:19]
                    )
                    if now > trial_end:
                        row['status'] = 'expired'
                        status = 'expired'
                        updated = True
                        logger.info(
                            f'[ASSISTS] Trial expired: '
                            f'{email}'
                        )
                except Exception:
                    pass

        # Deduplication — keep most recent
        # active or trial record per email
        if email in seen_emails:
            existing = seen_emails[email]
            existing_status = existing.get(
                'status', ''
            )
            # Current row is active/trial and
            # existing is expired/cancelled
            # — replace with current
            if (
                status in ['active', 'trial']
                and existing_status not in
                ['active', 'trial']
            ):
                seen_emails[email] = row
        else:
            seen_emails[email] = row

    # Build final deduped list
    all_rows = list(seen_emails.values())

    # Write back if statuses changed
    if updated:
        with open(FILE, 'w', newline='') as f:
            writer = csv.DictWriter(
                f, fieldnames=FIELDS
            )
            writer.writeheader()
            for row in all_rows:
                clean = {
                    k: row.get(k, '')
                    for k in FIELDS
                }
                writer.writerow(clean)

    # Return only active and valid trials
    for row in all_rows:
        if row.get('status') in [
            'active', 'trial'
        ]:
            active.append(row)

    logger.info(
        f'[ASSISTS] Active subscribers: '
        f'{len(active)}'
    )
    return active


=== FIX 4: SCHEDULER MISFIRE GUARD ===

In mora_assists.py add this function:

def check_missed_send_today():
    from datetime import datetime
    import pytz

    et = pytz.timezone('America/New_York')
    now = datetime.now(et)

    # Only check between 10:31 AM and 11:59 PM
    if now.hour < 10 or (
        now.hour == 10 and now.minute < 31
    ):
        return

    flag = (
        f'/var/data/picks_sent_'
        f'{now.strftime("%Y-%m-%d")}.flag'
    )

    if os.path.exists(flag):
        logger.info(
            '[RECOVERY] Picks already sent today'
        )
        return

    logger.warning(
        '[RECOVERY] Picks not sent — '
        'running recovery now'
    )
    run_daily_assists()


In app.py background_initializer()
after scheduler starts add:

import threading
def _recovery_check():
    import time
    time.sleep(10)
    try:
        from mora_assists import (
            check_missed_send_today
        )
        check_missed_send_today()
    except Exception as e:
        logger.error(f'Recovery check error: {e}')

threading.Thread(
    target=_recovery_check,
    daemon=True
).start()
