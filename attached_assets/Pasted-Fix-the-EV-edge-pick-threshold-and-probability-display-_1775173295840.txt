# We can estimate fair prob quickly to pre-filter

# Use simple average as quick estimate
quick_probs = []
for b in valid_books:
    op = b.get("over_price")
    up = b.get("under_price")
    if op and up:
        def imp(o):
            return (100/(o+100) if o > 0 
                    else abs(o)/(abs(o)+100))
        raw_o = imp(op)
        raw_u = imp(up)
        total = raw_o + raw_u
        if total > 0:
            quick_probs.append(raw_o / total)
