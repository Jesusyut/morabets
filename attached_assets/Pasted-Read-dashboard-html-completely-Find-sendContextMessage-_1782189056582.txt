async function sendContextMessage() {
  const input = document.getElementById(
    'ce-input'
  );
  const btn = document.getElementById(
    'ce-send-btn'
  );
  const messages = document.getElementById(
    'ce-messages'
  );

  if (!input || !messages) {
    console.error('CE: DOM elements missing');
    return;
  }

  const text = input.value.trim();
  if (!text) return;

  const passcode = sessionStorage.getItem(
    'ce_passcode'
  ) || '';

  if (!passcode) {
    document.getElementById(
      'ce-input-area'
    ).style.display = 'none';
    document.getElementById(
      'ce-passcode-gate'
    ).style.display = 'block';
    return;
  }

  // Add user message bubble
  const userBubble = document.createElement('div');
  userBubble.style.cssText = `
    align-self: flex-end;
    background: rgba(76,187,23,0.15);
    border: 1px solid rgba(76,187,23,0.3);
    border-radius: 14px 14px 2px 14px;
    padding: 10px 13px;
    max-width: 90%;
  `;
  userBubble.innerHTML = `<p style="font-family:'Inter',sans-serif;font-size:13px;color:#ffffff;margin:0;line-height:1.5;">${text.replace(/</g,'&lt;')}</p>`;
  messages.appendChild(userBubble);

  // Add loading bubble
  const loadBubble = document.createElement('div');
  loadBubble.style.cssText = `
    align-self: flex-start;
    background: rgba(255,255,255,0.04);
    border-radius: 14px 14px 14px 2px;
    padding: 12px 14px;
    max-width: 90%;
  `;
  loadBubble.innerHTML = `<p style="font-family:'Inter',sans-serif;font-size:12px;color:#6B9E5A;margin:0;letter-spacing:1px;">Analyzing board...</p>`;
  messages.appendChild(loadBubble);
  messages.scrollTop = messages.scrollHeight;

  input.value = '';
  btn.disabled = true;
  btn.textContent = 'Analyzing...';

  try {
    const payload = {
      passcode: passcode,
      user_message: text
    };

    console.log('CE: sending to /api/context-edge', payload);

    const res = await fetch('/api/context-edge', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json'
      },
      body: JSON.stringify(payload)
    });

    console.log('CE: response status', res.status);

    const raw = await res.text();
    console.log('CE: raw response', raw.substring(0, 200));

    if (res.status === 403) {
      sessionStorage.removeItem('ce_passcode');
      _ceAuthed = false;
      loadBubble.innerHTML = `<p style="font-family:'Inter',sans-serif;font-size:12px;color:#EF4444;margin:0;">Session expired. Re-enter passcode.</p>`;
      document.getElementById('ce-input-area').style.display = 'none';
      document.getElementById('ce-passcode-gate').style.display = 'block';
      btn.disabled = false;
      btn.textContent = 'Analyze Context Edge →';
      return;
    }

    if (!res.ok) {
      loadBubble.innerHTML = `<p style="font-family:'Inter',sans-serif;font-size:12px;color:#EF4444;margin:0;">Server error ${res.status}. Try again.</p>`;
      btn.disabled = false;
      btn.textContent = 'Analyze Context Edge →';
      return;
    }

    let data;
    try {
      data = JSON.parse(raw);
    } catch(parseErr) {
      console.error('CE: JSON parse failed', parseErr, raw);
      loadBubble.innerHTML = `<p style="font-family:'Inter',sans-serif;font-size:12px;color:#EF4444;margin:0;">Parse error. Raw: ${raw.substring(0,100)}</p>`;
      btn.disabled = false;
      btn.textContent = 'Analyze Context Edge →';
      return;
    }

    console.log('CE: parsed data keys', Object.keys(data));

    const responseText = data.response ||
      data.error ||
      'No response field in data.';

    // Convert markdown to HTML
    const formatted = responseText
      .replace(/&/g, '&amp;')
      .replace(/</g, '&lt;')
      .replace(/\*\*(.*?)\*\*/g, '<strong>$1</strong>')
      .replace(/## (.*?)(<br>|$)/g, '<strong style="color:#4CBB17;font-size:14px;">$1</strong><br>')
      .replace(/⚡/g, '<span style="color:#4CBB17">⚡</span>')
      .replace(/\bPLAY\b/g, '<span style="color:#4CBB17;font-weight:700;">PLAY</span>')
      .replace(/\bSMALL PLAY\b/g, '<span style="color:#EAB308;font-weight:700;">SMALL PLAY</span>')
      .replace(/\bPASS\b/g, '<span style="color:#EF4444;font-weight:700;">PASS</span>')
      .replace(/\n/g, '<br>');

    loadBubble.innerHTML = `
      <div style="
        font-family: 'Inter', sans-serif;
        font-size: 13px;
        color: #9CA3AF;
        line-height: 1.7;
      ">${formatted}</div>
    `;

    messages.scrollTop = messages.scrollHeight;

  } catch(err) {
    console.error('CE: fetch exception', err.message, err.stack);
    loadBubble.innerHTML = `
      <p style="font-family:'Inter',sans-serif;font-size:12px;color:#EF4444;margin:0;">
        Error: ${err.message}
      </p>
    `;
  }

  btn.disabled = false;
  btn.textContent = 'Analyze Context Edge →';
}

Key changes in this version:
- Uses appendChild not innerHTML +=
  (innerHTML += destroys existing DOM
  nodes and their event listeners —
  this was likely causing the crash)
- Uses res.text() first then JSON.parse
  so parse errors are caught separately
- Every failure path shows the actual
  error message instead of generic text
- Console logs at every step for
  browser devtools debugging
- Passcode checked from sessionStorage
  directly on every call
- No reference to external variables
  that may not be in scope
