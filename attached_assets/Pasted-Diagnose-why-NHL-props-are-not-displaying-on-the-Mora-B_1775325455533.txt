--- FIX D: If /api/nhl/props route missing ---

Add to app.py:

@app.route("/api/nhl/props")
def nhl_props():
    try:
        import os, time
        from enrichment import load_props_from_file

        cache_file = "nhl_props_cache.json"
        cache_fresh = False

        if os.path.exists(cache_file):
            age = time.time() - os.path.getmtime(
                cache_file
            )
            cache_fresh = age < 82800  # 23 hours

        if cache_fresh:
            props = load_props_from_file(cache_file)
            if props:
                return jsonify({
                    "props":  props,
                    "count":  len(props),
                    "sport":  "NHL",
                    "cached": True,
                    "tiers": {
                        "LOCK": len([
                            p for p in props
                            if p.get(
                                "confidence_tier"
                            ) == "LOCK"
                        ]),
                        "FIRE": len([
                            p for p in props
                            if p.get(
                                "confidence_tier"
                            ) == "FIRE"
                        ]),
                        "LOW": len([
                            p for p in props
                            if p.get(
                                "confidence_tier"
                            ) == "LOW"
                        ])
                    }
                })

        props = _fetch_and_process_nhl_props()
        return jsonify({
            "props":  props,
            "count":  len(props),
            "sport":  "NHL",
            "cached": False
        })

    except Exception as e:
        logger.error(
            f"[NHL PROPS] Route error: {e}"
        )
        return jsonify({
            "error": str(e),
            "props": [],
            "count": 0
        }), 500

--- FIX E: If _fetch_and_process_nhl_props 
    missing or broken ---

Add or replace in app.py:

def _fetch_and_process_nhl_props():
    """Fetch, group, and cache NHL props."""
    try:
        from nhl_odds_api import fetch_nhl_props
        from odds_api import group_props_by_player
        from enrichment import cache_props_to_file

        logger.info("[NHL PROPS] Fetching...")
        raw = fetch_nhl_props()

        if not raw:
            logger.warning(
                "[NHL PROPS] No raw props — "
                "checking stale cache"
            )
            from enrichment import load_props_from_file
            stale = load_props_from_file(
                "nhl_props_cache.json"
            )
            if stale:
                logger.info(
                    f"[NHL PROPS] Serving "
                    f"{len(stale)} stale props"
                )
                return stale
            return []

        props = group_props_by_player(raw)

        if not props:
            logger.warning(
                "[NHL PROPS] 0 after grouping"
            )
            return []

        cache_props_to_file(
            props, "nhl_props_cache.json"
        )
        logger.info(
            f"[NHL PROPS] {len(props)} cached ✅"
        )
        return props

    except Exception as e:
        logger.error(
            f"[NHL PROPS] Failed: {e}",
            exc_info=True
        )
        return []

In dashboard.html find the NHL tab fetch.

NOT:

  /api/nhl/environment (old endpoint)
  /api/nhl/props?key=... (auth removed, no key needed)

The fetch should look exactly like:

async function loadSportBoard(sport) {
  try {
    showLoadingSkeleton();
    
    const endpoint = sport === 'nfl'
      ? `/api/${sport}/props`
      : `/api/${sport}/props`;
    
    const res = await fetch(
      `${endpoint}?t=${Date.now()}`
    );
    const data = await res.json();

    const props = (
      data.props || 
      data.picks || 
      []
    );

    console.log(
      `[${sport.toUpperCase()}] ` +
      `${props.length} props received`
    );

    if (props.length === 0) {
      showEmptyState(sport);
      return;
    }

    renderBoard(props, [], {}, sport);

  } catch(err) {
    console.error('[BOARD]', err);
    showErrorState(sport);
  }
}
