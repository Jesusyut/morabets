=== CHANGE 4: ADD COUNTDOWN TIMER JS ===

Add this function to the JavaScript
section of dashboard.html.
Runs independently of popup show logic.
Starts immediately on page load.

World Cup first match kicks off:
June 11 2026 at 8:00 PM ET UTC-4
= June 12 2026 00:00:00 UTC

function startWorldCupCountdown() {
  var target = new Date(
    'June 12, 2026 00:00:00 UTC'
  ).getTime();

  var timer = setInterval(function() {
    var now = Date.now();
    var diff = target - now;

    if (diff <= 0) {
      clearInterval(timer);
      var el = document.getElementById(
        'wc-countdown'
      );
      if (el) el.style.display = 'none';
      return;
    }

    var days = Math.floor(
      diff / (1000 * 60 * 60 * 24)
    );
    var hours = Math.floor(
      (diff % (1000 * 60 * 60 * 24)) /
      (1000 * 60 * 60)
    );
    var mins = Math.floor(
      (diff % (1000 * 60 * 60)) /
      (1000 * 60)
    );

    var d = document.getElementById('wc-days');
    var h = document.getElementById('wc-hours');
    var m = document.getElementById('wc-mins');

    if (d) d.textContent = days;
    if (h) h.textContent = hours;
    if (m) m.textContent = mins;

  }, 1000);
}

startWorldCupCountdown();

async function submitNewsletter() {
  var emailEl = document.getElementById(
    'newsletter-email-input'
  );
  var email = emailEl
    ? emailEl.value.trim()
    : '';

  if (!email || !email.includes('@')) {
    if (emailEl) {
      emailEl.style.borderColor = '#EF4444';
      emailEl.placeholder =
        'Enter a valid email';
    }
    return;
  }

  var btn = document.querySelector(
    '[onclick="submitNewsletter()"]'
  );
  if (btn) {
    btn.textContent = 'Sending...';
    btn.disabled = true;
  }

  try {
    await fetch(
      'https://hooks.zapier.com/hooks/catch/22797974/uj2zhly/',
      {
        method: 'POST',
        mode: 'no-cors',
        headers: {
          'Content-Type': 'application/json'
        },
        body: JSON.stringify({
          email: email,
          source: 'newsletter_popup',
          timestamp: new Date().toISOString()
        })
      }
    );
  } catch(e) {
    console.log('Webhook error:', e);
  }

  if (emailEl) emailEl.style.display = 'none';
  if (btn) btn.style.display = 'none';

  var success = document.getElementById(
    'newsletter-success'
  );
  if (success) {
    success.style.display = 'block';
  }

  localStorage.setItem(
    'mora_newsletter_signed_up',
    'true'
  );

  setTimeout(function() {
    closeNewsletterPopup();
  }, 3000);
}

NOTE: mode: 'no-cors' is required for
Zapier webhooks from the browser.
Without it the browser blocks the request
even though the data goes through.
This is the correct production pattern
for all Zapier browser fetches.
