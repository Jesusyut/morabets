--- FIX A: If scheduler is not running ---

The most likely cause is Render restarted
the service and the background thread
that starts the scheduler did not
re-register the mora_assists_daily job.

Shell commands and manual test runs
in the Render shell do NOT affect the
scheduler — they run in a separate
process. The scheduler lives in the
main gunicorn process only.

Every time Render restarts (deploys,
sleep, crash) the scheduler must
re-register all jobs on startup.

In app.py find background_initializer().

Replace the entire function with:

def background_initializer():
    """
    Registers all scheduled jobs
    and primes caches on startup.
    Runs in a background thread so
    it does not block server startup.
    """
    global app_initialized
    import time
    time.sleep(5)

    try:
        logger.info(
            "🚀 Starting background init..."
        )

        # ── START SCHEDULER ──────────────────
        if not scheduler.running:
            scheduler.start()
            logger.info("✅ Scheduler started")
        else:
            logger.info(
                "⚠️ Scheduler already running"
            )

        # ── REGISTER MORA ASSISTS JOB ─────────
        # This MUST run every startup because
        # APScheduler does not persist jobs
        # across process restarts on Render.
        try:
            from mora_assists import run_daily_assists

            scheduler.add_job(
                func=run_daily_assists,
                trigger='cron',
                hour=10,
                minute=30,
                timezone='America/New_York',
                id='mora_assists_daily',
                name='Mora Assists 10:30AM ET',
                replace_existing=True,
                misfire_grace_time=3600,
                coalesce=True
            )
            logger.info(
                "✅ mora_assists_daily "
                "job registered"
            )

            # Log when it will next fire
            job = scheduler.get_job(
                'mora_assists_daily'
            )
            if job:
                logger.info(
                    f"⏰ Next send: "
                    f"{job.next_run_time}"
                )

        except Exception as e:
            logger.error(
                f"❌ Failed to register "
                f"mora_assists job: {e}"
            )

        # ── REGISTER MLB PROPS JOB ────────────
        try:
            scheduler.add_job(
                func=update_player_props,
                trigger='cron',
                hour=10,
                minute=0,
                timezone='America/New_York',
                id='update_mlb_props_daily',
                name='MLB Props 10AM ET',
                replace_existing=True,
                misfire_grace_time=3600,
                coalesce=True
            )
            logger.info(
                "✅ MLB props job registered"
            )
        except Exception as e:
            logger.warning(
                f"MLB props job error: {e}"
            )

        # ── REGISTER NHL PROPS JOB ────────────
        try:
            from nhl_odds_api import (
                fetch_nhl_props
            )
            scheduler.add_job(
                func=update_nhl_props,
                trigger='cron',
                hour=10,
                minute=5,
                timezone='America/New_York',
                id='update_nhl_props_daily',
                name='NHL Props 10:05AM ET',
                replace_existing=True,
                misfire_grace_time=3600,
                coalesce=True
            )
            logger.info(
                "✅ NHL props job registered"
            )
        except Exception as e:
            logger.warning(
                f"NHL props job error: {e}"
            )

        # ── REGISTER ODDS REFRESH ─────────────
        try:
            scheduler.add_job(
                func=update_odds,
                trigger='cron',
                hour='10,18',
                minute=10,
                timezone='America/New_York',
                id='update_game_odds',
                name='Game Odds Refresh',
                replace_existing=True,
                misfire_grace_time=3600,
                coalesce=True
            )
            logger.info(
                "✅ Game odds job registered"
            )
        except Exception as e:
            logger.warning(
                f"Odds job error: {e}"
            )

        # ── LOG ALL REGISTERED JOBS ───────────
        all_jobs = scheduler.get_jobs()
        logger.info(
            f"📋 Total scheduled jobs: "
            f"{len(all_jobs)}"
        )
        for j in all_jobs:
            logger.info(
                f"  {j.id} → {j.next_run_time}"
            )

        # ── PRIME CACHES ─────────────────────
        logger.info("🔄 Priming caches...")
        try:
            update_odds()
            logger.info("✅ Odds cache primed")
        except Exception as e:
            logger.warning(
                f"Odds cache failed: {e}"
            )

        try:
            update_player_props()
            logger.info("✅ MLB props primed")
        except Exception as e:
            logger.warning(
                f"MLB props failed: {e}"
            )

        try:
            update_nhl_props()
            logger.info("✅ NHL props primed")
        except Exception as e:
            logger.warning(
                f"NHL props failed: {e}"
            )

        app_initialized = True
        logger.info(
            "🎉 Background init complete"
        )

    except Exception as e:
        logger.error(
            f"Background init failed: {e}",
            exc_info=True
        )
        app_initialized = True

@app.route('/api/scheduler-status')
def scheduler_status():
    """
    Check scheduler health and 
    registered jobs.
    Visit: morabets.com/api/scheduler-status
    """
    try:
        from datetime import datetime
        jobs = scheduler.get_jobs()

        mora_job = scheduler.get_job(
            'mora_assists_daily'
        )

        return jsonify({
            'scheduler_running':
                scheduler.running,
            'total_jobs':
                len(jobs),
            'mora_assists_daily': {
                'registered':
                    mora_job is not None,
                'next_run':
                    str(mora_job.next_run_time)
                    if mora_job else 'NOT REGISTERED',
                'misfire_grace':
                    mora_job.misfire_grace_time
                    if mora_job else None
            },
            'all_jobs': [
                {
                    'id': j.id,
                    'next_run': str(
                        j.next_run_time
                    )
                }
                for j in jobs
            ],
            'server_time_utc':
                str(datetime.utcnow()),
            'status':
                '✅ healthy'
                if mora_job else
                '❌ mora_assists_daily missing'
        })
    except Exception as e:
        return jsonify({
            'error': str(e),
            'status': '❌ error'
        }), 500

@app.route(
    '/api/send-picks-now',
    methods=['POST']
)
def send_picks_now():
    """
    Manually trigger Mora Assists send.
    POST to this route to fire picks
    immediately to all active subscribers.
    
    Use when scheduler misses a day.
    """
    try:
        from mora_assists import run_daily_assists
        import threading

        def run_in_background():
            logger.info(
                "[MANUAL] Picks send triggered"
            )
            result = run_daily_assists()
            logger.info(
                f"[MANUAL] Result: {result}"
            )

        thread = threading.Thread(
            target=run_in_background,
            daemon=True
        )
        thread.start()

        return jsonify({
            'status': 'triggered',
            'message':
                'Picks send started in '
                'background. Check Render '
                'logs for results.',
            'check_logs': True
        })

    except Exception as e:
        return jsonify({
            'error': str(e)
        }), 500
