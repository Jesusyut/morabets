=== PROBLEM 4: NEW SORT ORDER ===

Current sort: LOCK → FIRE → LOW (tier only)

New sort: highest no_vig_prob first AND within same 
probability, best payout (least negative / most positive 
American odds) rises to top.

This means a 63% prop at -108 ranks ABOVE a 63% prop 
at -145. Both have same probability but -108 pays better.

Replace the existing sort with this comparator in JS 
(applied to uniqueProps before rendering):

uniqueProps.sort((a, b) => {
  // Primary: highest no_vig_prob first
  if (b.no_vig_prob !== a.no_vig_prob) {
    return b.no_vig_prob - a.no_vig_prob;
  }
  // Secondary: best payout (highest American odds value)
  // -108 > -145, +110 > -108
  return b.best_over_price - a.best_over_price;
});

Also update sort_props_by_tier() in probability.py to

=== PROBLEM 6: MONEYLINES AS FALLBACK ===

When a sport tab loads and player props are sparse or 
all fail the two-sided check, fall back to showing 
game moneylines (h2h) as picks instead of an empty board.

For moneylines the no-vig calc works perfectly since
every game has both home and away prices.

In the tab load function for each sport:

async function loadSportProps(sport) {
  const props = await fetchProps(sport);
  const validProps = filterAndDedup(props);
  
  if (validProps.length < 5) {
    // Not enough props — load moneylines instead
    const odds = await fetchOdds(sport);
    const mlPicks = convertMoneylinesToPicks(odds);
    renderCards(mlPicks);
  } else {
    renderCards(validProps);
  }
}

function convertMoneylinesToPicks(oddsData) {
  const picks = [];
  for (const game of oddsData) {
    for (const book of (game.bookmakers || [])) {
      const h2h = book.markets?.find(m => m.key === 'h2h');
      if (!h2h) continue;
      
      const outcomes = h2h.outcomes;
      if (outcomes.length < 2) continue;
      
      const side1 = outcomes[0];
      const side2 = outcomes[1];
      
      // Calculate no-vig for both sides
      function toImplied(o) {
        return o > 0 ? 100/(o+100) : Math.abs(o)/(Math.abs(o)+100);
      }
      const imp1 = toImplied(side1.price);
      const imp2 = toImplied(side2.price);
      const total = imp1 + imp2;
      const prob1 = Math.round((imp1/total)*100*10)/10;
      const prob2 = Math.round((imp2/total)*100*10)/10;
      
      function getTier(p) {
        return p >= 70 ? 'LOCK' : p >= 50 ? 'FIRE' : 'LOW';
      }
      
      picks.push({
        player: side1.name,       // team name as "player"
        stat: 'Moneyline',
        stat_label: 'Moneyline Win',
        line: null,
        no_vig_prob: prob1,
        confidence_tier: getTier(prob1),
        best_over_price: side1.price,
        best_book: book.title,
        all_books: [{
          book: book.title,
          over_price: side1.price,
          under_price: side2.price
        }],
        game_time: game.commence_time,
        sport: game.sport_key?.includes('nhl') ? 'NHL' :
               game.sport_key?.includes('mlb') ? 'MLB' : 'NFL'
      });
      
      picks.push({
        player: side2.name,
        stat: 'Moneyline',
        stat_label: 'Moneyline Win',
        line: null,
        no_vig_prob: prob2,
        confidence_tier: getTier(prob2),
        best_over_price: side2.price,
        best_book: book.title,
        all_books: [{
          book: book.title,
          over_price: side2.price,
          under_price: side1.price
        }],
        game_time: game.commence_time,
        sport: game.sport_key?.includes('nhl') ? 'NHL' :
               game.sport_key?.includes('mlb') ? 'MLB' : 'NFL'
      });
      
      break; // one book per game is enough for fallback
    }
  }
  
  // Dedup and sort same as props
  const seen = new Set();
  return picks
    .filter(p => {
      if (seen.has(p.player)) return false;
      seen.add(p.player);
      return true;
    })
    .sort((a, b) => {
      if (b.no_vig_prob !== a.no_vig_prob)
        return b.no_vig_prob - a.no_vig_prob;
      return b.best_over_price - a.best_over_price;
    });
}
