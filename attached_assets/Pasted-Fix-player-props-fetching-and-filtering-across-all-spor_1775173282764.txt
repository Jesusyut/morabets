def fetch_player_props():
    """
    Fetch MLB player props for all today's events.
    
    Per Odds API docs:
    - Step 1: GET /sports/baseball_mlb/events (FREE)
    - Step 2: GET /sports/baseball_mlb/events/{id}/odds
              one event at a time, markets batched
    
    Player name is in outcome["description"]
    Side (Over/Under) is in outcome["name"]
    
    Returns list of props with BOTH over_price 
    and under_price paired per player per book.
    One-sided props (over only) are included 
    with under_price=None.
    """
    now = datetime.utcnow()
    future = now + timedelta(hours=24)
    start_time = now.strftime('%Y-%m-%dT%H:%M:%SZ')
    end_time = future.strftime('%Y-%m-%dT%H:%M:%SZ')

    if not ODDS_API_KEY:
        print("[ERROR] ODDS_API_KEY is not set")
        return []

    # Step 1: Fetch events (FREE — no quota cost)
    try:
        event_resp = requests.get(
            f"{BASE_URL}/sports/baseball_mlb/events",
            params={
                "apiKey":           ODDS_API_KEY,
                "commenceTimeFrom": start_time,
                "commenceTimeTo":   end_time,
                "dateFormat":       "iso"
            },
            timeout=20
        )
        event_resp.raise_for_status()
        events = event_resp.json()
        print(f"[PROPS] Found {len(events)} MLB events")
    except Exception as e:
        print(f"[ERROR] Failed to fetch MLB events: {e}")
        return []

    if not events:
        print("[PROPS] No MLB events today")
        return []

    # MLB prop markets — batched to respect quota
    # Cost: len(markets_in_batch) x 1 region per event
    MARKET_BATCHES = [
        [
            "batter_hits",
            "batter_total_bases",
            "batter_home_runs"
        ],
        [
            "pitcher_strikeouts",
            "pitcher_hits_allowed",
            "batter_rbis"
        ]
    ]

    # Expand book list for better two-sided coverage
    # More books = more chances to find both Over+Under
    BOOKS = (
        "draftkings,fanduel,betmgm,"
        "caesars,pointsbetus,betrivers"
    )

    all_props = []

    for event in events:
        eid        = event.get("id")
        home_team  = event.get("home_team", "")
        away_team  = event.get("away_team", "")
        game_time  = event.get("commence_time", "")
        matchup    = f"{away_team} @ {home_team}"

        if not eid:
            continue

        for batch_idx, markets in enumerate(MARKET_BATCHES):
            # Space requests to avoid 429 rate limit
            if batch_idx > 0:
                time.sleep(0.5)

            try:
                resp = requests.get(
                    f"{BASE_URL}/sports/baseball_mlb"
                    f"/events/{eid}/odds",
                    params={
                        "apiKey":      ODDS_API_KEY,
                        "regions":     "us",
                        "markets":     ",".join(markets),
                        "oddsFormat":  "american",
                        "bookmakers":  BOOKS
                    },
                    timeout=20
                )

                # 422 = no props for this event yet
                if resp.status_code == 422:
                    print(
                        f"[PROPS] No props yet: {matchup}"
                    )
                    continue

                # 429 = rate limited
                if resp.status_code == 429:
                    print("[PROPS] Rate limited — pausing")
                    time.sleep(3)
                    continue

                resp.raise_for_status()
                data = resp.json()

                # Log quota
                remaining = resp.headers.get(
                    "x-requests-remaining", "?"
                )
                print(
                    f"[PROPS] {matchup} batch "
                    f"{batch_idx}: quota={remaining}"
                )

                for book in data.get("bookmakers", []):
                    book_title = book.get("title", "")
                    if not book_title:
                        continue

                    for market in book.get("markets", []):
                        market_key = market.get("key", "")
                        outcomes   = market.get(
                            "outcomes", []
                        )

                        # === CORRECT PAIRING LOGIC ===
                        # Per official docs:
                        # outcome["description"] = player name
                        # outcome["name"] = "Over" or "Under"
                        # Group by player within this market+book

                        player_map = {}

                        for outcome in outcomes:
                            # Player name is in description
                            player_name = outcome.get(
                                "description", ""
                            )
                            side  = outcome.get(
                                "name", ""
                            )  # "Over" or "Under"
                            price = outcome.get("price")
                            point = outcome.get("point")

                            if not player_name:
                                continue
                            if price is None:
                                continue

                            if player_name not in player_map:
                                player_map[player_name] = {
                                    "over_price":  None,
                                    "under_price": None,
                                    "line":        point
                                }

                            if side == "Over":
                                player_map[player_name][
                                    "over_price"
                                ] = price
                                player_map[player_name][
                                    "line"
                                ] = point
                            elif side == "Under":
                                player_map[player_name][
                                    "under_price"
                                ] = price

                        # Build paired prop per player
                        for player_name, sides in (
                            player_map.items()
                        ):
                            over_p  = sides["over_price"]
                            under_p = sides["under_price"]
                            line    = sides["line"]

                            # Must have at least over price
                            if over_p is None:
                                continue

                            all_props.append({
                                "player":      player_name,
                                "stat":        market_key,
                                "line":        line,
                                "over_price":  over_p,
                                "under_price": under_p,
                                "odds":        over_p,
                                "bookmaker":   book_title,
                                "home_team":   home_team,
                                "away_team":   away_team,
                                "matchup":     matchup,
                                "game_time":   game_time,
                                "sport":       "MLB"
                            })

            except requests.exceptions.HTTPError as e:
                print(
                    f"[ERROR] HTTP {e.response.status_code}"
                    f" for {matchup} batch {batch_idx}"
                )
                continue
            except Exception as e:
                print(
                    f"[ERROR] Props fetch failed for "
                    f"{matchup} batch {batch_idx}: {e}"
                )
                continue

    print(
        f"[PROPS] Collected {len(all_props)} raw props "
        f"from {len(events)} events"
    )

    # Stat breakdown for debugging
    stat_counts = {}
    for p in all_props:
        k = p.get("stat", "unknown")
        stat_counts[k] = stat_counts.get(k, 0) + 1
    print(f"[PROPS] By market: {stat_counts}")

    return all_props

def group_props_by_player(props):
    """
    Group raw props by player+stat+line.
    Collect all books per group.
    Calculate no-vig or implied probability.
    Apply minimum probability filter.
    Sort by tier then probability.
    """
    from probability import (
        no_vig_probability,
        get_confidence_tier,
        implied_probability_single,
        get_confidence_tier_implied,
        sort_props_by_tier
    )

    groups = {}

    for p in props:
        player  = p.get("player", "")
        stat    = p.get("stat", "")
        line    = p.get("line")
        book    = p.get("bookmaker", "").lower().strip()

        if not player or not stat:
            continue

        key = f"{player}_{stat}_{line}"

        if key not in groups:
            groups[key] = {
                "player":    player,
                "stat":      stat,
                "stat_label": get_stat_display_name(stat),
                "line":      line,
                "home_team": p.get("home_team", ""),
                "away_team": p.get("away_team", ""),
                "matchup":   p.get("matchup", ""),
                "game_time": p.get("game_time", ""),
                "sport":     p.get("sport", "MLB"),
                "all_books": []
            }

        # Handle both old shape (odds only) and 
        # new shape (over_price + under_price)
        over_p  = p.get("over_price") or p.get("odds")
        under_p = p.get("under_price")

        if over_p is not None:
            groups[key]["all_books"].append({
                "book":        book,
                "over_price":  over_p,
                "under_price": under_p
            })

    result = []

    for key, group in groups.items():
        books = group["all_books"]
        if not books:
            continue

        # Separate two-sided from one-sided
        two_sided = [
            b for b in books
            if b.get("over_price") is not None
            and b.get("under_price") is not None
        ]
        one_sided = [
            b for b in books
            if b.get("over_price") is not None
            and b.get("under_price") is None
        ]

        if two_sided:
            # ── FULL NO-VIG CALCULATION ──────────────
            # Use sharpest book (lowest total vig)
            # for fair price, best odds for betting line

            def vig_total(b):
                def imp(o):
                    return (100/(o+100) if o > 0
                            else abs(o)/(abs(o)+100))
                return (imp(b["over_price"]) +
                        imp(b["under_price"]))

            sharpest = min(two_sided, key=vig_total)
            best     = max(
                two_sided,
                key=lambda b: b["over_price"]
            )
            worst    = min(
                two_sided,
                key=lambda b: b["over_price"]
            )

            nv_prob = no_vig_probability(
                sharpest["over_price"],
                sharpest["under_price"]
            )

            # ── HARD FILTER: minimum 52% no-vig ─────
            # Below 52% = market says this side is
            # the underdog — do not surface it
            if nv_prob < 52.0:
                continue

            def to_imp(o):
                return (100/(o+100) if o > 0
                        else abs(o)/(abs(o)+100))

            savings = round(
                (to_imp(worst["over_price"]) -
                 to_imp(best["over_price"])) * 100, 1
            )

            result.append({
                **group,
                "all_books":        two_sided,
                "no_vig_prob":      nv_prob,
                "prob_type":        "no_vig",
                "prob_label":       "True Prob",
                "confidence_tier":  get_confidence_tier(
                    nv_prob
                ),
                "best_book":        best["book"],
                "best_over_price":  best["over_price"],
                "worst_over_price": worst["over_price"],
                "line_shop_savings": savings,
                "implied_only":     False,
                "surfaced":         True,
                "enriched":         True
            })

        elif one_sided:
            # ── IMPLIED PROBABILITY ONLY ─────────────
            best = max(
                one_sided,
                key=lambda b: b["over_price"]
            )
            over_p = best["over_price"]

            implied_prob = implied_probability_single(
                over_p
            )

            # ── HARD FILTER: minimum 58% implied ────
            # Implied includes vig so 58% implied
            # ≈ 52-54% true probability after removal.
            # Below 58% = not worth showing casually.
            # This eliminates all the 40% dogs.
            if implied_prob < 58.0:
                continue

            result.append({
                **group,
                "all_books":        one_sided,
                "no_vig_prob":      implied_prob,
                "prob_type":        "implied",
                "prob_label":       "Implied",
                "confidence_tier":  get_confidence_tier_implied(
                    implied_prob
                ),
                "best_book":        best["book"],
                "best_over_price":  over_p,
                "worst_over_price": over_p,
                "line_shop_savings": 0,
                "implied_only":     True,
                "surfaced":         True,
                "enriched":         True
            })

        else:
            continue

    print(
        f"[PROPS] Grouped {len(props)} raw props → "
        f"{len(result)} unique props after filtering"
    )

    return sort_props_by_tier(result)

def implied_probability_single(over_price: int) -> float:
    """
    Raw implied probability from one side only.
    No vig removal possible — labeled as implied.
    Returns as percentage e.g. 58.4
    """
    try:
        if over_price > 0:
            raw = 100 / (over_price + 100)
        else:
            raw = abs(over_price) / (
                abs(over_price) + 100
            )
        return round(raw * 100, 1)
    except Exception:
        return 50.0


def get_confidence_tier_implied(prob: float) -> str:
    """
    Tier for implied-only props.
    Higher thresholds since implied includes vig.
    58% implied ≈ 52% no-vig after vig removal.
    """
    if prob >= 70:
        return "LOCK"
    elif prob >= 58:
        return "FIRE"
    else:
        return "LOW"


def no_vig_probability(
    over_price: int, 
    under_price: int
) -> float:
    """
    Strip vig and return true probability for 
    the OVER side as a percentage e.g. 63.2
    """
    def to_implied(american):
        if american > 0:
            return 100 / (american + 100)
        else:
            return abs(american) / (
                abs(american) + 100
            )
    try:
        raw_over  = to_implied(over_price)
        raw_under = to_implied(under_price)
        total = raw_over + raw_under
        if total == 0:
            return 50.0
        return round((raw_over / total) * 100, 1)
    except Exception:
        return 50.0


def get_confidence_tier(no_vig_prob: float) -> str:
    if no_vig_prob >= 70:
        return "LOCK"
    elif no_vig_prob >= 55:
        return "FIRE"
    else:
        return "LOW"


def sort_props_by_tier(props: list) -> list:
    TIER_ORDER = {"LOCK": 0, "FIRE": 1, 
                  "LOW": 2, "EDGE": 1}
    return sorted(
        props,
        key=lambda p: (
            TIER_ORDER.get(
                p.get("confidence_tier", "LOW"), 2
            ),
            -p.get("no_vig_prob", 0),
            -p.get("best_over_price", -999)
        )
    )
