## STEP 4 — BUILD THE ACCESS TOKEN SYSTEM

If generate_access_token does not exist, add to app.py:

```python
import random
import string
import json
import os
from datetime import datetime

TOKENS_FILE = 'access_tokens.json'

def load_tokens():
    """Load all access tokens from persistent file."""
    if os.path.exists(TOKENS_FILE):
        try:
            with open(TOKENS_FILE, 'r') as f:
                return json.load(f)
        except:
            return {}
    return {}

def save_tokens(tokens):
    """Save all access tokens to persistent file."""
    try:
        with open(TOKENS_FILE, 'w') as f:
            json.dump(tokens, f, indent=2)
    except Exception as e:
        print(f'[ERROR] Could not save tokens: {e}')

def generate_access_token(customer_name, customer_email):
    """
    Generate unique access token.
    Format: MB-[INITIALS]-[4chars]-[4chars]
    One token per email — duplicate emails get same token.
    """
    # Check if email already has a token (enforce 1 per user)
    tokens = load_tokens()
    for t, data in tokens.items():
        if data.get('email') == customer_email and data.get('active'):
            print(f'[TOKEN] Existing token found for {customer_email}')
            return t
    
    # Extract initials from name
    if customer_name and customer_name.strip():
        parts = customer_name.strip().split()
        if len(parts) >= 2:
            initials = (parts[0][0] + parts[-1][0]).upper()
        else:
            initials = (parts[0][:2]).upper()
    else:
        # Fallback: use first two chars of email username
        initials = customer_email.split('@')[0][:2].upper()
    
    # Generate unique random segments
    chars = string.ascii_uppercase + string.digits
    while True:
        seg1 = ''.join(random.choices(chars, k=4))
        seg2 = ''.join(random.choices(chars, k=4))
        token = f"MB-{initials}-{seg1}-{seg2}"
        # Ensure uniqueness
        if token not in tokens:
            break
    
    print(f'[TOKEN] Generated {token} for {customer_email}')
    return token

def get_token_by_email(email):
    """Look up active token for a given email."""
    tokens = load_tokens()
    for t, data in tokens.items():
        if data.get('email') == email and data.get('active'):
            return t
    return None

def deactivate_token_by_subscription(subscription_id):
    """Deactivate token when subscription is cancelled."""
    tokens = load_tokens()
    for t, data in tokens.items():
        if data.get('subscription_id') == subscription_id:
            tokens[t]['active'] = False
            tokens[t]['deactivated_at'] = datetime.utcnow().isoformat()
            save_tokens(tokens)
            print(f'[TOKEN] Deactivated {t} (subscription cancelled)')
            return t
    return None

def validate_token(token):
    """Check if a token is valid and active. Used by dashboard gate."""
    tokens = load_tokens()
    if token in tokens:
        return tokens[token].get('active', False)
    return False
```

## STEP 7 — UPDATE THE WEBHOOK HANDLER

Find the existing /webhook route. Ensure it:

```python
@app.route('/webhook', methods=['POST'])
def stripe_webhook():
    payload = request.get_data()
    sig_header = request.headers.get('Stripe-Signature')
    webhook_secret = os.environ.get('STRIPE_WEBHOOK_SECRET')
    
    try:
        event = stripe.Webhook.construct_event(payload, sig_header, webhook_secret)
    except stripe.error.SignatureVerificationError as e:
        print(f'[WEBHOOK] Signature failed: {e}')
        return jsonify({'error': 'Invalid signature'}), 400
    except Exception as e:
        print(f'[WEBHOOK] Error: {e}')
        return jsonify({'error': str(e)}), 400
    
    print(f'[WEBHOOK] Event received: {event["type"]}')
    
    # Payment completed — generate and deliver access key
    if event['type'] == 'checkout.session.completed':
        session = event['data']['object']
        customer_email = session.get('customer_details', {}).get('email', '')
        customer_name = session.get('customer_details', {}).get('name', '') or ''
        subscription_id = session.get('subscription', '')
        customer_id = session.get('customer', '')
        
        if customer_email:
            tokens = load_tokens()
            
            # Check for existing active token (1 per user enforcement)
            existing = get_token_by_email(customer_email)
            if existing:
                print(f'[WEBHOOK] Token already exists for {customer_email}: {existing}')
                # Update subscription ID if missing
                if subscription_id and not tokens[existing].get('subscription_id'):
                    tokens[existing]['subscription_id'] = subscription_id
                    save_tokens(tokens)
            else:
                # Generate new token
                token = generate_access_token(customer_name, customer_email)
                tokens[token] = {
                    'email': customer_email,
                    'name': customer_name,
                    'stripe_customer_id': customer_id,
                    'subscription_id': subscription_id,
                    'created_at': datetime.utcnow().isoformat(),
                    'active': True
                }
                save_tokens(tokens)
                
                # Send email with access key
                send_access_key_email(customer_email, customer_name, token)
                print(f'[WEBHOOK] New subscriber processed: {customer_email}')
    
    # Subscription cancelled — deactivate token
    elif event['type'] == 'customer.subscription.deleted':
        subscription = event['data']['object']
        sub_id = subscription.get('id')
        deactivated = deactivate_token_by_subscription(sub_id)
        if deactivated:
            print(f'[WEBHOOK] Token deactivated for subscription {sub_id}')
    
    # Payment failed — log it
    elif event['type'] == 'invoice.payment_failed':
        invoice = event['data']['object']
        customer_email = invoice.get('customer_email', 'unknown')
        print(f'[WEBHOOK] Payment failed for {customer_email}')
    
    return jsonify({'status': 'ok'}), 200
```

@app.route('/admin/health')
def admin_health():
    """Quick health check — shows token count and system status."""
    # Basic protection — check for admin key
    admin_key = request.args.get('key')
    if admin_key != os.environ.get('ADMIN_KEY', 'mora-admin-2026'):
        return jsonify({'error': 'unauthorized'}), 401
    
    tokens = load_tokens()
    active = [t for t, d in tokens.items() if d.get('active')]
    inactive = [t for t, d in tokens.items() if not d.get('active')]
    
    return jsonify({
        'status': 'running',
        'total_tokens': len(tokens),
